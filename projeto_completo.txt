"""Bulk market ingestion.

The critical property: Steam does not always honour the requested `count`.
It commonly returns 10 rows regardless. Advancing `start` by the REQUESTED
size then skips everything Steam did not return -- a silent data-loss bug.
Pagination must step by the number of rows actually received.
"""
from __future__ import annotations

import time

import httpx
import pytest

from tradeup.prices.bulk import REQUESTED_PAGE_SIZE, BulkMarket


def _catalogue(n: int) -> list[str]:
    return [f"Item {i:04d} (Field-Tested)" for i in range(n)]


def _server(names: list[str], rows_per_page: int):
    """A fake Steam that ignores `count` and returns `rows_per_page`."""

    class Fake:
        calls = 0

        def get(self, url, params=None):
            Fake.calls += 1
            q = (params.get("query") or "").lower()
            hits = [n for n in names if q in n.lower()] if q else names
            start = params["start"]
            rows = hits[start:start + rows_per_page]
            return httpx.Response(200, json={
                "success": True, "total_count": len(hits),
                "results": [{"hash_name": n, "sell_price": 100 + i,
                             "sell_listings": 5} for i, n in enumerate(rows)]},
                request=httpx.Request("GET", url))

    return Fake()


def test_no_items_are_skipped_when_steam_caps_the_page(tmp_path):
    """Regression: this is the bug that silently lost ~90% of the market."""
    names = _catalogue(250)
    m = BulkMarket(currency="BRL", cache_path=tmp_path / "c.sqlite", page_delay=0)
    m.client = _server(names, rows_per_page=10)   # asked 100, gets 10

    notes: list[str] = []
    stats = m.sweep(progress=notes.append)

    assert stats.effective_page_size == 10
    assert stats.items == 250
    missing = [n for n in names if m.cache.get(n, 7, 99999) is None]
    assert not missing, f"{len(missing)} items skipped, e.g. {missing[:3]}"
    assert stats.pages == 25          # 250/10, not 250/100
    assert any("Steam gave 10" in n for n in notes)


def test_full_page_size_is_used_when_honoured(tmp_path):
    names = _catalogue(250)
    m = BulkMarket(currency="BRL", cache_path=tmp_path / "f.sqlite", page_delay=0)
    m.client = _server(names, rows_per_page=REQUESTED_PAGE_SIZE)
    stats = m.sweep()
    assert stats.effective_page_size == 100
    assert stats.pages == 3
    assert stats.items == 250
    assert not [n for n in names if m.cache.get(n, 7, 99999) is None]


def test_row_count_counts_unpriced_rows_too(tmp_path):
    """A row with no usable price still occupies a catalogue position."""
    page = {"success": True, "total_count": 3, "results": [
        {"hash_name": "A (FT)", "sell_price": 100, "sell_listings": 1},
        {"hash_name": "B (FT)", "sell_price": None, "sell_price_text": ""},
        {"hash_name": "C (FT)", "sell_price": 0},
    ]}
    assert BulkMarket.row_count(page) == 3          # step start by 3
    assert len(list(BulkMarket.parse_results(page))) == 1   # but keep 1


def test_parse_results_handles_real_payload_shapes():
    page = {"success": True, "results": [
        {"name": "AK-47 | Elite Build",
         "hash_name": "AK-47 | Elite Build (Field-Tested)",
         "sell_price": 341, "sell_price_text": "R$ 3,41", "sell_listings": 1420},
        {"hash_name": "Souvenir MP9 | Storm (Minimal Wear)",
         "sell_price": 89, "sell_listings": 37},
        {"name": "Fallback | Text (Factory New)", "sell_price": None,
         "sell_price_text": "R$ 12,50", "sell_listings": 9},
        "not-a-dict",
    ]}
    rows = list(BulkMarket.parse_results(page))
    assert ("AK-47 | Elite Build (Field-Tested)", 3.41, 1420) in rows
    assert ("Souvenir MP9 | Storm (Minimal Wear)", 0.89, 37) in rows
    assert ("Fallback | Text (Factory New)", 12.50, 9) in rows
    assert len(rows) == 3


def test_targeted_queries_cost_far_fewer_requests(tmp_path):
    names = ([f"AK-47 | Skin {i} (FT)" for i in range(25)]
             + [f"MP9 | Skin {i} (FT)" for i in range(200)])
    names.sort()

    m = BulkMarket(currency="BRL", cache_path=tmp_path / "q.sqlite", page_delay=0)
    m.client = _server(names, rows_per_page=10)
    stats = m.sweep_queries(["AK-47"])

    assert stats.written == 25
    assert stats.pages == 3                       # vs 23 for a full sweep
    assert stats.queries_done == 1
    assert m.cache.get("AK-47 | Skin 7 (FT)", 7, 99999) is not None


def test_wanted_filter_restricts_writes_not_paging(tmp_path):
    names = _catalogue(100)
    want = {names[i] for i in range(0, 100, 10)}
    m = BulkMarket(currency="BRL", cache_path=tmp_path / "w.sqlite", page_delay=0)
    m.client = _server(names, rows_per_page=10)
    stats = m.sweep(wanted=want)
    assert stats.items == 100          # paged through everything
    assert stats.written == len(want)  # kept only what was asked for


def test_sweep_recovers_from_429(tmp_path):
    calls = {"n": 0}

    class Limited:
        def get(self, url, params=None):
            calls["n"] += 1
            if calls["n"] <= 2:
                return httpx.Response(429, headers={"Retry-After": "1"}, text="",
                                      request=httpx.Request("GET", url))
            return httpx.Response(200, json={
                "success": True, "total_count": 1,
                "results": [{"hash_name": "OK | Item (FN)", "sell_price": 500,
                             "sell_listings": 10}]},
                request=httpx.Request("GET", url))

    m = BulkMarket(currency="BRL", cache_path=tmp_path / "r.sqlite",
                   page_delay=0, max_backoff=2)
    m.client = Limited()
    t0 = time.monotonic()
    stats = m.sweep()
    assert stats.rate_limited == 2 and stats.written == 1
    assert time.monotonic() - t0 < 30      # backoff is capped


def test_sweep_stops_on_non_json_body(tmp_path):
    """Steam serves HTML during outages; do not loop forever on it."""

    class Html:
        def get(self, url, params=None):
            return httpx.Response(200, text="<html>maintenance</html>",
                                  request=httpx.Request("GET", url))

    m = BulkMarket(currency="BRL", cache_path=tmp_path / "h.sqlite",
                   page_delay=0, max_retries=2, max_backoff=1)
    m.client = Html()
    stats = m.sweep()
    assert stats.written == 0 and stats.errors >= 1


def test_empty_page_terminates_the_sweep(tmp_path):
    class Empty:
        def get(self, url, params=None):
            return httpx.Response(200, json={"success": True, "total_count": 999,
                                             "results": []},
                                  request=httpx.Request("GET", url))

    m = BulkMarket(currency="BRL", cache_path=tmp_path / "e.sqlite", page_delay=0)
    m.client = Empty()
    stats = m.sweep()
    assert stats.pages == 1 and stats.written == 0

    # buyer pays 11.50 -> seller receives 10.00
    assert steam_net_from_buyer_price(11.50) == pytest.approx(10.00, abs=0.01)

    assert steam_net_from_buyer_price(10.00) > 8.50

def test_normalization_depends_on_the_skins_own_range():

# --- odds -------------------------------------------------------------
def _pool_factory(pools):
    return lambda coll, rarity: pools.get((coll, int(rarity)), [])


def test_single_collection_is_uniform_under_both_models():
    outs = [mk(f"O{i}", "A", Rarity.RESTRICTED, 0, 1) for i in range(4)]
    inp = mk("I", "A", Rarity.MILSPEC, 0, 1)
    pool = _pool_factory({("A", int(Rarity.RESTRICTED)): outs})
    items = [InputItem(inp, 0.1) for _ in range(10)]
    for model in ("outcome_weighted", "per_input"):
        p = outcome_probabilities(items, pool, model)
        assert all(v == pytest.approx(0.25) for v in p.values())


def test_models_diverge_on_mixed_collections():
    a_out = [mk("A1", "A", Rarity.RESTRICTED, 0, 1),
             mk("A2", "A", Rarity.RESTRICTED, 0, 1)]
    b_out = [mk(f"B{i}", "B", Rarity.RESTRICTED, 0, 1) for i in range(4)]
    pool = _pool_factory({("A", int(Rarity.RESTRICTED)): a_out,
                          ("B", int(Rarity.RESTRICTED)): b_out})
    items = ([InputItem(mk("IA", "A", Rarity.MILSPEC, 0, 1), 0.1)] * 5 +
             [InputItem(mk("IB", "B", Rarity.MILSPEC, 0, 1), 0.1)] * 5)

    ow = outcome_probabilities(items, pool, "outcome_weighted")
    assert sum(ow[s.id] for s in a_out) == pytest.approx(1 / 3)

    pi = outcome_probabilities(items, pool, "per_input")
    assert sum(pi[s.id] for s in a_out) == pytest.approx(0.5)
    assert pi["A1|A"] == pytest.approx(0.25)

    for p in (ow, pi):
        assert sum(p.values()) == pytest.approx(1.0)


def test_dead_end_collection_contributes_no_outcome_mass():
    outs = [mk("A1", "A", Rarity.RESTRICTED, 0, 1)]
    pool = _pool_factory({("A", int(Rarity.RESTRICTED)): outs})
    items = ([InputItem(mk("IA", "A", Rarity.MILSPEC, 0, 1), 0.1)] * 2 +
             [InputItem(mk("IB", "B", Rarity.MILSPEC, 0, 1), 0.1)] * 8)
    p = outcome_probabilities(items, pool, "per_input")
    assert p == {"A1|A": pytest.approx(1.0)}


# --- full contract ----------------------------------------------------
def test_evaluate_contract_end_to_end():
    outs = [mk("Cheap", "A", Rarity.RESTRICTED, 0.00, 0.50),
            mk("Jackpot", "A", Rarity.RESTRICTED, 0.00, 0.50)]
    inp = mk("In", "A", Rarity.MILSPEC, 0.00, 1.00)
    pool = _pool_factory({("A", int(Rarity.RESTRICTED)): outs})
    items = [InputItem(inp, 0.05, price=1.00) for _ in range(10)]
    prices = {"Cheap": 5.0, "Jackpot": 40.0}
    res = evaluate_contract(items, pool, lambda s, w, st: prices.get(s.name))

    assert res.cost == pytest.approx(10.0)
    assert res.avg_normalized == pytest.approx(0.05)
    assert res.outcomes[0].wear == "Factory New"
    expected = (0.5 * steam_net_from_buyer_price(5.0)
                + 0.5 * steam_net_from_buyer_price(40.0))
    assert res.ev_net == pytest.approx(expected, abs=0.01)
    assert res.profit_chance == pytest.approx(0.5)
    assert res.unpriced_mass == 0.0


def test_unpriced_outcomes_are_tracked_not_silently_zeroed():
    outs = [mk("Known", "A", Rarity.RESTRICTED, 0, 0.5),

    res = evaluate_contract(
        items, pool, lambda s, w, st: 10.0 if s.name == "Known" else None)
    assert res.unpriced_mass == pytest.approx(0.5)

"""Souvenir trade-up support (Valve rule change, 22 May 2026).

Souvenir items became valid contract inputs; souvenir attributes are stripped
and the output is ALWAYS normal quality. StatTrak still cannot mix with
Souvenir. Most Consumer/Industrial supply is souvenir-bearing, so excluding
it silently emptied the two cheapest tiers.
"""
import json

import pytest

from tradeup.data.loader import SkinDB, derive_qualities, parse_skins_payload
from tradeup.engine import evaluate_contract, validate_contract
from tradeup.models import InputItem, Rarity, Skin
from tradeup.optimizer import build_units
from tradeup.prices.steam import PriceBook
from tradeup.pricing import PriceService, PricingPolicy


def mk(name, coll, rarity, mn=0.0, mx=1.0, sv=True, st=False):
    return Skin(f"{name}|{coll}", name, rarity, coll, mn, mx, st, sv)


def test_market_hash_name_prefixes():
    s = mk("AK-47 | Redline", "C", Rarity.MILSPEC)
    assert s.market_hash_name("Field-Tested") == "AK-47 | Redline (Field-Tested)"
    assert s.market_hash_name("Field-Tested", souvenir=True) == \
        "Souvenir AK-47 | Redline (Field-Tested)"
    assert s.market_hash_name("Field-Tested", stattrak=True) == \
        "StatTrak\u2122 AK-47 | Redline (Field-Tested)"
    # mutually exclusive: StatTrak wins, never "StatTrak Souvenir"
    n = s.market_hash_name("Field-Tested", stattrak=True, souvenir=True)
    assert n.startswith("StatTrak") and "Souvenir" not in n


def test_souvenir_collections_are_not_filtered_out():
    """Regression: the loader used to drop any collection named *Souvenir*."""
    payload = [
        {"name": "MAG-7 | Counter Terrace", "min_float": 0.0, "max_float": 1.0,
         "rarity": {"name": "Consumer Grade"}, "souvenir": True, "stattrak": False,
         "crates": [{"name": "Cologne 2014 Cobblestone Souvenir Package"}],
         "collections": [{"name": "The Souvenir Cobblestone Collection"}]},
        {"name": "Five-SeveN | Kami", "min_float": 0.0, "max_float": 0.6,
         "rarity": {"name": "Industrial Grade"}, "souvenir": True, "stattrak": False,
         "crates": [{"name": "Cologne 2014 Cobblestone Souvenir Package"}],
         "collections": [{"name": "The Souvenir Cobblestone Collection"}]},
    ]
    db = parse_skins_payload(payload)
    assert len(db) == 2
    assert db.tradeable_collections(Rarity.CONSUMER) == \
        ["The Souvenir Cobblestone Collection"]
    assert db.skins[0].souvenir_capable is True


def test_souvenir_inputs_do_not_change_odds_float_or_output_price():
    """A souvenir input is identical to a normal one except for what it cost."""
    out = mk("Out", "A", Rarity.INDUSTRIAL, 0.0, 0.5, sv=False)
    inp = mk("In", "A", Rarity.CONSUMER, 0.0, 1.0)
    pool = lambda c, r: [out] if (c, int(r)) == ("A", int(Rarity.INDUSTRIAL)) else []

    normal = [InputItem(inp, 0.05, 1.0, souvenir=False) for _ in range(10)]
    souv = [InputItem(inp, 0.05, 0.5, souvenir=True) for _ in range(10)]

    seen = []

    def price_fn(skin, wear, st):
        seen.append((skin.name, wear, st))
        return 20.0

    a = evaluate_contract(normal, pool, price_fn)
    b = evaluate_contract(souv, pool, price_fn)

    assert a.avg_normalized == b.avg_normalized
    assert [o.probability for o in a.outcomes] == [o.probability for o in b.outcomes]
    assert a.ev_net == b.ev_net          # output priced identically
    assert b.cost < a.cost               # only the input cost differs
    assert b.profit > a.profit
    assert b.n_souvenir_inputs == 10
    assert all(st is False for _, _, st in seen)


def test_output_is_never_priced_as_souvenir_even_if_all_inputs_are():
    """Trade-up output is always NORMAL quality (Valve, 22 May 2026)."""
    from tradeup.prices.steam import Quote

    out = mk("Out", "A", Rarity.INDUSTRIAL, 0.0, 0.5, sv=True)
    inp = mk("In", "A", Rarity.CONSUMER, 0.0, 1.0, sv=True)
    pool = lambda c, r: [out] if (c, int(r)) == ("A", int(Rarity.INDUSTRIAL)) else []

    asked: list[str] = []

    class Src:
        def get(self, name):
            asked.append(name)
            return Quote(name, 9.0, 10.0, 100, 0.0)

    ps = PriceService(Src(), PricingPolicy(min_volume=1, haircut=0.0))
    items = [InputItem(inp, 0.05, 0.1, souvenir=True) for _ in range(10)]
    evaluate_contract(items, pool, ps.price_fn())

    assert asked, "price_fn was never called"
    assert all(not n.startswith("Souvenir ") for n in asked), \
        f"output was priced as a Souvenir: {asked}"


def test_mixing_souvenir_and_normal_is_legal_but_stattrak_is_not():
    s = mk("A", "C", Rarity.CONSUMER)
    st_skin = mk("B", "C", Rarity.CONSUMER, sv=False, st=True)
    mixed = [InputItem(s, 0.1, souvenir=i % 2 == 0) for i in range(10)]
    assert validate_contract(mixed) == []

    bad = mixed[:9] + [InputItem(st_skin, 0.1, stattrak=True)]
    assert any("StatTrak" in p for p in validate_contract(bad))

    wrong = [InputItem(s, 0.1, souvenir=True),
             InputItem(mk("C", "C", Rarity.MILSPEC), 0.1)]
    assert any("multiple rarities" in p for p in validate_contract(wrong))

    no_sv = mk("D", "C", Rarity.CONSUMER, sv=False)
    assert any("no Souvenir variant" in p
               for p in validate_contract([InputItem(no_sv, 0.1, souvenir=True)]))


def test_build_units_adds_souvenir_variants(tmp_path):
    skin = mk("MAG-7 | Counter Terrace", "Coll", Rarity.CONSUMER, 0.0, 1.0)
    out = mk("Out", "Coll", Rarity.INDUSTRIAL, 0.0, 0.5, sv=False)

    class DB:
        def tradeable_collections(self, r): return ["Coll"]
        def pool(self, c, r): return [skin] if r == Rarity.CONSUMER else [out]
        def by_id(self, i): return None

    snap = tmp_path / "p.json"
    snap.write_text(json.dumps({
        "MAG-7 | Counter Terrace (Field-Tested)":
            {"lowest": 1.00, "median": 1.10, "volume": 500},
        "Souvenir MAG-7 | Counter Terrace (Field-Tested)":
            {"lowest": 0.40, "median": 0.45, "volume": 300},
    }), encoding="utf-8")
    ps = PriceService(PriceBook.load(snap), PricingPolicy(min_input_volume=10))

    off = build_units(DB(), ps, Rarity.CONSUMER, souvenir=False)
    on = build_units(DB(), ps, Rarity.CONSUMER, souvenir=True)
    assert len(on["Coll"]) == len(off["Coll"]) + 1
    cheapest = min(on["Coll"], key=lambda u: u.price)
    assert cheapest.souvenir is True and cheapest.price == 0.40
    assert cheapest.label.startswith("Souvenir ")

    # StatTrak contracts must never pull in souvenir units
    assert build_units(DB(), ps, Rarity.CONSUMER, souvenir=True, stattrak=True) == {}

# --- quality derivation (crate-based, not the blanket flag) -----------
def test_souvenir_capability_comes_from_crates_not_the_flag():
    """Regression: upstream sets `souvenir: true` on ~every finish."""
    # case-distributed finish: StatTrak yes, Souvenir no -- even though the
    # upstream payload claims souvenir=True
    assert derive_qualities({
        "stattrak": True, "souvenir": True,
        "crates": [{"name": "Fracture Case"}]}) == (True, False)

    # tournament package finish: Souvenir yes, StatTrak no
    assert derive_qualities({
        "stattrak": True, "souvenir": True,
        "crates": [{"name": "Cologne 2014 Cobblestone Souvenir Package"}]}) \
        == (False, True)

    # appears in both a case and a souvenir package -> both allowed
    assert derive_qualities({"crates": [
        {"name": "Fracture Case"},
        {"name": "Berlin 2019 Dust II Souvenir Package"}]}) == (True, True)

    # no crate data -> fall back to flags, but never claim both at once
    assert derive_qualities(
        {"stattrak": True, "souvenir": True, "crates": []}) == (True, False)
    assert derive_qualities({"stattrak": False, "souvenir": True}) == (False, True)

"""Skin/collection database loading and indexing.

Source (free, no API key, community maintained):
    https://raw.githubusercontent.com/ByMykel/CSGO-API/main/public/api/en/skins.json

The older `bymykel.github.io/CSGO-API/api/...` host no longer serves JSON --
it answers with an HTML error page, which a bare `.json()` turns into a
confusing "Expecting value" JSONDecodeError. Mirrors are tried in order and
the body is validated before parsing.

ENCODING: every read/write pins `encoding="utf-8"`. Skin names contain CJK
and other non-Latin characters and Windows defaults text I/O to cp1252,
which cannot encode them. Never drop the explicit encoding.

SOUVENIR: since the 22 May 2026 update Souvenir items are valid trade-up
inputs, so Souvenir collections are NOT filtered out -- most Consumer and
Industrial supply lives in tournament/map collections.

Do NOT trust the upstream `souvenir` boolean on its own: in the current
payload it is set on essentially every finish, which would claim a
"Souvenir <skin>" listing exists for all ~1400 entries. The structural
signal is the CRATE: a souvenir variant exists where the skin is
distributed through a "... Souvenir Package". `souvenir_capable` is derived
from that, with the flag used only as a fallback when crate data is missing.
The market is the final arbiter -- a unit is only built if a "Souvenir ..."
listing actually returns a price.
"""
from __future__ import annotations

import json
from dataclasses import asdict
from pathlib import Path
from typing import Iterable, Optional, Union

from ..models import Rarity, Skin

CSGO_API_MIRRORS: tuple[str, ...] = (
    "https://raw.githubusercontent.com/ByMykel/CSGO-API/main/public/api/en/skins.json",
    "https://cdn.jsdelivr.net/gh/ByMykel/CSGO-API@main/public/api/en/skins.json",
    "https://bymykel.github.io/CSGO-API/api/en/skins.json",  # legacy, usually dead
)
CSGO_API_SKINS = CSGO_API_MIRRORS[0]

ROOT = Path(__file__).resolve().parents[2]
DATA_DIR = ROOT / "data"
DB_PATH = DATA_DIR / "skins.json"
SAMPLE_PATH = DATA_DIR / "sample_skins.json"

SOUVENIR_CRATE_HINT = "souvenir package"


class SyncError(RuntimeError):
    """Raised with actionable guidance when the skin DB cannot be downloaded."""


def _crate_names(item: dict) -> list[str]:
    return [(c.get("name") or "")
            for c in (item.get("crates") or []) if isinstance(c, dict)]


def derive_qualities(item: dict) -> tuple[bool, bool]:
    """(stattrak_capable, souvenir_capable) for one CSGO-API entry.

    Souvenir and StatTrak are mutually exclusive in CS2: StatTrak comes from
    weapon cases, Souvenir from tournament packages, and no item is both.
    Crate membership tells us which distribution a finish belongs to, and is
    far more reliable than the upstream booleans.
    """
    crates = _crate_names(item)
    if crates:
        souvenir = any(SOUVENIR_CRATE_HINT in c.lower() for c in crates)
        stattrak = any(SOUVENIR_CRATE_HINT not in c.lower() for c in crates)
        if souvenir and stattrak:
            # appears in both a case and a souvenir package: allow both
            return True, True
        return stattrak, souvenir
    # No crate data (many map-collection finishes). Fall back to the flags,
    # but never claim both -- that combination does not exist in game.
    st = bool(item.get("stattrak", False))
    sv = bool(item.get("souvenir", False))
    if st and sv:
        return True, False
    return st, sv


class SkinDB:
    """In-memory index over the skin universe."""

    def __init__(self, skins: Iterable[Skin]):
        self.skins: list[Skin] = list(skins)
        self._by_id: dict[str, Skin] = {s.id: s for s in self.skins}
        self._by_name: dict[str, Skin] = {s.name.lower(): s for s in self.skins}
        self._by_coll_rarity: dict[tuple[str, int], list[Skin]] = {}
        for s in self.skins:
            self._by_coll_rarity.setdefault((s.collection, int(s.rarity)), []).append(s)

    # -- lookups ----------------------------------------------------------
    def by_id(self, sid: str) -> Optional[Skin]:
        return self._by_id.get(sid)

    def by_name(self, name: str) -> Optional[Skin]:
        return self._by_name.get(name.lower())

    def pool(self, collection: str, rarity: Rarity) -> list[Skin]:
        """Every skin of `rarity` inside `collection` -- the outcome pool."""
        return self._by_coll_rarity.get((collection, int(rarity)), [])

    def collections(self, rarity: Optional[Rarity] = None) -> list[str]:
        if rarity is None:
            return sorted({s.collection for s in self.skins})
        return sorted({c for (c, r) in self._by_coll_rarity if r == int(rarity)})

    def tradeable_collections(self, input_rarity: Rarity) -> list[str]:
        """Collections with BOTH inputs at `input_rarity` and outputs above."""
        nxt = input_rarity.next_tier()
        if nxt is None:
            return []
        return [c for c in self.collections(input_rarity) if self.pool(c, nxt)]

    def __len__(self) -> int:
        return len(self.skins)

    def quality_sanity(self) -> list[str]:
        """Warn when derived quality flags look implausible.

        A near-100% souvenir rate means the upstream boolean was trusted
        blindly; souvenir variants only exist for tournament/map collections,
        so a realistic share is well under half the universe.
        """
        warnings: list[str] = []
        n = len(self.skins)
        if not n:
            return ["database is empty"]
        sv = sum(1 for s in self.skins if s.souvenir_capable)
        both = sum(1 for s in self.skins if s.souvenir_capable and s.stattrak_capable)
        if sv / n > 0.90:
            warnings.append(
                f"{sv}/{n} ({sv/n:.0%}) skins marked Souvenir-capable. That is "
                "almost certainly wrong -- Souvenir variants only exist for "
                "tournament/map collections.")
        if both / max(n, 1) > 0.50:
            warnings.append(
                f"{both}/{n} skins marked both StatTrak- and Souvenir-capable. "
                "These are mutually exclusive for a given drop source.")
        return warnings

    # -- persistence ------------------------------------------------------
    def save(self, path: Union[str, Path] = DB_PATH) -> Path:
        path = Path(path)
        path.parent.mkdir(parents=True, exist_ok=True)
        payload = []
        for s in self.skins:
            d = asdict(s)
            d["rarity"] = int(s.rarity)
            payload.append(d)
        # encoding is mandatory: skin names include non-cp1252 characters
        path.write_text(
            json.dumps(payload, ensure_ascii=False, indent=1), encoding="utf-8")
        return path

    @classmethod
    def load(cls, path: Optional[Union[str, Path]] = None) -> "SkinDB":
        p = Path(path) if path else (DB_PATH if DB_PATH.exists() else SAMPLE_PATH)
        raw = json.loads(Path(p).read_text(encoding="utf-8"))
        return cls(
            Skin(
                id=d["id"], name=d["name"], rarity=Rarity(int(d["rarity"])),
                collection=d["collection"],
                min_float=float(d["min_float"]), max_float=float(d["max_float"]),
                stattrak_capable=bool(d.get("stattrak_capable", True)),
                # tolerate databases written before souvenir support
                souvenir_capable=bool(
                    d.get("souvenir_capable", d.get("souvenir_only", False))),
            )
            for d in raw
        )

def _fetch_json(url: str, timeout: float) -> list:
    """GET a URL and return parsed JSON, with real diagnostics on failure."""
    import httpx

    headers = {
        "User-Agent": "cs2-tradeup-lab/1.0 (+https://github.com/ByMykel/CSGO-API)",
        "Accept": "application/json, text/plain, */*",
    }
    with httpx.Client(timeout=timeout, follow_redirects=True, headers=headers) as c:
        r = c.get(url)

    if r.status_code != 200:
        raise SyncError(f"HTTP {r.status_code} from {url}")
    body = r.text.lstrip()
    if not body:
        raise SyncError(f"Empty body from {url}")
    if body[0] not in "[{":
        # Almost always an HTML 404/redirect page. Show the start of it so the
        # cause is obvious instead of a bare "Expecting value".
        ctype = r.headers.get("content-type", "?")
        raise SyncError(f"{url} returned {ctype}, not JSON. First bytes: {body[:120]!r}")
    try:
        data = r.json()
    except ValueError as e:
        raise SyncError(f"{url} returned malformed JSON: {e}") from e
    if not isinstance(data, list) or not data:
        raise SyncError(f"{url} returned an unexpected payload shape")
    return data


def fetch_skins_payload(url: Optional[str] = None, timeout: float = 60.0) -> list:
    """Fetch the raw skins payload, trying each mirror in turn."""
    urls = [url] if url else list(CSGO_API_MIRRORS)
    problems: list[str] = []
    for u in urls:
        try:
            return _fetch_json(u, timeout)
        except SyncError as e:
            problems.append(str(e))
        except Exception as e:  # network/DNS/TLS
            problems.append(f"{u}: {type(e).__name__}: {e}")
    raise SyncError(
        "Could not download the skin database. Tried:\n  - " + "\n  - ".join(problems)
        + "\n\nWorkaround: download the file manually in a browser from\n  "
        + CSGO_API_MIRRORS[0]
        + "\nthen run:  python -m tradeup.cli sync-db --from-file skins.json")


def parse_skins_payload(raw: list) -> SkinDB:
    """Normalize a CSGO-API `skins.json` payload into a SkinDB."""
    skins: list[Skin] = []
    seen: set[str] = set()
    for item in raw:
        if not isinstance(item, dict):
            continue
        try:
            rarity = Rarity.parse((item.get("rarity") or {}).get("name", ""))
        except ValueError:
            continue
        colls = item.get("collections") or []
        if not colls:
            continue  # case-only / no collection -> cannot be a trade-up input
        name = item.get("name") or ""
        # CSGO-API names include the wear in some entries; strip it
        if name.endswith(")") and "(" in name:
            name = name[: name.rfind("(")].strip()
        mn, mx = item.get("min_float"), item.get("max_float")
        if mn is None or mx is None:
            continue
        st_cap, sv_cap = derive_qualities(item)
        for coll in colls:
            cname = coll.get("name") or ""
            if not cname:
                continue
            sid = f"{name}|{cname}"
            if sid in seen:
                continue
            seen.add(sid)
            skins.append(Skin(
                id=sid, name=name, rarity=rarity, collection=cname,
                min_float=float(mn), max_float=float(mx),
                stattrak_capable=st_cap, souvenir_capable=sv_cap,
            ))
    if not skins:
        raise SyncError(
            "Payload downloaded but no usable skins were parsed -- the upstream "
            "schema likely changed (expected: name, rarity, collections, "
            "min_float, max_float).")
    return SkinDB(skins)

"""Bulk price ingestion via Steam's paginated market search.

`/market/priceoverview/` returns ONE item per request. A tier sweep is
thousands of names against an endpoint whose rate limit is enforced per IP
and whose ban can last hours (and re-arms while you are banned). You cannot
win that race by tuning the delay.

`/market/search/render/?norender=1` returns a page of items per request --
the same endpoint the market web page itself calls.

IMPORTANT -- PAGE SIZE IS NOT GUARANTEED
Steam does not always honour `count`. It commonly caps the page at 10 rows
regardless of what you ask for, and will silently degrade under load. The
first version of this module assumed it got what it asked for and advanced
`start` by the REQUESTED size, which skipped every item between the rows it
actually received. That is a data-correctness bug, not just a slow sweep.

So: `start` always advances by the number of rows actually parsed, the
effective page size is detected from the first page, and the ETA is
recomputed from that. If Steam gives 10 rows per request, a full-market
sweep is ~3.5k requests and is NOT worth it -- use `queries=` to fetch only
the weapons you care about, or fall back to per-item pricing for a
shortlist.

Trade-off vs priceoverview, stated plainly:
  * search/render gives `sell_price` (lowest listing) and `sell_listings`
    (how many are listed right now).
  * priceoverview gives `lowest_price`, `median_price` and `volume`
    (units SOLD in 24h).
Bulk has no true median and no sales volume; `sell_listings` is a supply
count, not a liquidity measure. Confirm a shortlist before spending money.
"""
from __future__ import annotations

import random
import time
from dataclasses import dataclass, field
from pathlib import Path
from typing import Callable, Iterable, Iterator, Optional, Union

import httpx

from .steam import CURRENCY, ClientState, PriceCache, Quote, parse_price

APPID_CS2 = 730
SEARCH_URL = "https://steamcommunity.com/market/search/render/"

# What we ask for. Steam frequently gives less -- see the module docstring.
REQUESTED_PAGE_SIZE = 100

# If the effective page size is at or below this, a full sweep is impractical
# and the caller should be told to narrow the search instead.
SMALL_PAGE_WARN = 20


@dataclass(slots=True)
class BulkStats:
    pages: int = 0
    items: int = 0
    written: int = 0
    retries: int = 0
    rate_limited: int = 0
    errors: int = 0
    effective_page_size: Optional[int] = None
    total_count: Optional[int] = None
    queries_done: int = 0
    short_pages: int = 0          # pages smaller than the detected size

    @property
    def estimated_pages_remaining(self) -> Optional[int]:
        if not self.effective_page_size or not self.total_count:
            return None
        seen = self.items
        left = max(self.total_count - seen, 0)
        return -(-left // self.effective_page_size)


class BulkMarket:
    """Paginated Steam market reader.

    `sweep()` walks the whole catalogue. `sweep_queries()` runs one paginated
    search per query string, which is dramatically cheaper when you only need
    a few weapons and Steam is capping the page size.
    """

    def __init__(
        self,
        currency: str = "USD",
        cache_path: Union[str, Path] = Path("data/prices.sqlite"),
        page_delay: float = 6.0,
        timeout: float = 30.0,
        max_retries: int = 6,
        max_backoff: float = 300.0,
        page_size: int = REQUESTED_PAGE_SIZE,
    ):
        self.currency = CURRENCY.get(currency.upper(), 1)
        self.cache = PriceCache(cache_path)
        self.page_delay = page_delay
        self.max_retries = max_retries
        self.max_backoff = max_backoff
        self.page_size = page_size
        self.state = ClientState()
        self.stats = BulkStats()
        self.client = httpx.Client(
            timeout=httpx.Timeout(timeout, connect=10.0),
            headers={
                "User-Agent": (
                    "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                    "AppleWebKit/537.36 (KHTML, like Gecko) "
                    "Chrome/126.0 Safari/537.36"),
                "Accept": "application/json, text/javascript, */*; q=0.01",
                "Accept-Language": "en-US,en;q=0.9",
                "Referer": "https://steamcommunity.com/market/",
            },
        )

    def close(self) -> None:
        self.client.close()

    # -- one page ---------------------------------------------------------
    def _page(self, start: int, query: str = "") -> Optional[dict]:
        backoff = min(15.0, self.max_backoff)
        for attempt in range(self.max_retries):
            self.state.set("requesting", f"start={start} attempt {attempt + 1}")
            try:
                r = self.client.get(SEARCH_URL, params={
                    "query": query,
                    "start": start,
                    "count": self.page_size,
                    "search_descriptions": 0,
                    "sort_column": "name",
                    "sort_dir": "asc",
                    "appid": APPID_CS2,
                    "norender": 1,
                    "currency": self.currency,
                })
            except httpx.HTTPError as e:
                self.stats.errors += 1
                self.stats.retries += 1
                self.state.set("backoff", f"{type(e).__name__}, {backoff:.0f}s")
                time.sleep(backoff)
                backoff = min(backoff * 2, self.max_backoff)
                continue

            if r.status_code == 429:
                self.stats.rate_limited += 1
                self.stats.retries += 1
                ra = float(r.headers.get("Retry-After", 0) or 0)
                wait = min(max(ra, backoff), self.max_backoff)
                self.state.set("backoff", f"429, waiting {wait:.0f}s")
                time.sleep(wait + random.uniform(0, 3))
                backoff = min(backoff * 2, self.max_backoff)
                continue

            if r.status_code != 200:
                self.stats.errors += 1
                self.stats.retries += 1
                self.state.set("backoff", f"HTTP {r.status_code}, {backoff:.0f}s")
                time.sleep(backoff)
                backoff = min(backoff * 2, self.max_backoff)
                continue

            try:
                data = r.json()
            except ValueError:
                self.stats.errors += 1
                self.stats.retries += 1
                self.state.set("backoff", f"non-JSON body, {backoff:.0f}s")
                time.sleep(backoff)
                backoff = min(backoff * 2, self.max_backoff)
                continue

            if not data.get("success"):
                return None
            return data
        return None

    # -- parsing ----------------------------------------------------------
    @staticmethod
    def parse_results(data: dict) -> Iterator[tuple[str, float, int]]:
        """Yield (market_hash_name, price, listing_count) from a page.

        `sell_price` is an integer in cents of the requested currency;
        `sell_price_text` is the localized string used as a fallback.
        """
        for row in data.get("results") or []:
            if not isinstance(row, dict):
                continue
            name = row.get("hash_name") or row.get("name")
            if not name:
                continue
            cents = row.get("sell_price")
            if cents is None:
                p = parse_price(row.get("sell_price_text") or "")
                if p is None:
                    continue
                cents = int(round(p * 100))
            try:
                price = float(cents) / 100.0
            except (TypeError, ValueError):
                continue
            if price <= 0:
                continue
            try:
                listings = int(row.get("sell_listings") or 0)
            except (TypeError, ValueError):
                listings = 0
            yield name, price, listings

    @staticmethod
    def row_count(data: dict) -> int:
        """How many rows the page actually contained (parsed or not).

        Distinct from the number we keep: a row without a usable price is
        still a position in the catalogue that `start` must step over.
        """
        rows = data.get("results")
        return len(rows) if isinstance(rows, list) else 0

    # -- pagination core --------------------------------------------------
    def _paginate(
        self,
        query: str,
        wanted: Optional[set[str]],
        max_pages: int,
        progress: Optional[Callable[[str], None]],
        label: str = "",
    ) -> None:
        start = 0
        total: Optional[int] = None
        pages_here = 0

        while True:
            data = self._page(start, query)
            if data is None:
                break
            self.stats.pages += 1
            pages_here += 1

            if total is None:
                try:
                    total = int(data.get("total_count") or 0)
                except (TypeError, ValueError):
                    total = 0
                if self.stats.total_count is None:
                    self.stats.total_count = total

            n_rows = self.row_count(data)
            if n_rows == 0:
                break

            # Detect the real page size from the first full-looking page.
            if self.stats.effective_page_size is None:
                self.stats.effective_page_size = n_rows
                if n_rows < self.page_size and progress:
                    progress(
                        f"NOTE: asked for {self.page_size} rows/page, Steam gave "
                        f"{n_rows}. Paging by {n_rows}."
                        + (f" A full sweep of {total} items needs about "
                           f"{-(-total // n_rows)} requests."
                           if total and n_rows <= SMALL_PAGE_WARN else ""))
            elif n_rows < self.stats.effective_page_size:
                self.stats.short_pages += 1

            for name, price, listings in self.parse_results(data):
                self.stats.items += 1
                if wanted is not None and name not in wanted:
                    continue
                # `listings` is SUPPLY, not 24h sales. Stored in the volume
                # column so downstream policy has a liquidity proxy, but it
                # is deliberately weaker: see the module docstring.
                self.cache.put(
                    Quote(name, price, None, listings, time.time()), self.currency)
                self.stats.written += 1

            if progress:
                left = self.stats.estimated_pages_remaining
                progress(
                    f"{label}page {self.stats.pages} start={start}"
                    f"{f'/{total}' if total else ''} rows={n_rows} "
                    f"kept={self.stats.written} 429s={self.stats.rate_limited}"
                    + (f" ~{left} pages left" if left else ""))

            # CRITICAL: advance by what we actually received. Advancing by the
            # requested size skips every item Steam did not return.
            start += n_rows

            if total and start >= total:
                break
            if max_pages and pages_here >= max_pages:
                break

            if self.page_delay > 0:
                self.state.set("throttling", f"page delay {self.page_delay:.0f}s")
                time.sleep(self.page_delay
                           + random.uniform(0, self.page_delay * 0.25))

    # -- public sweeps ----------------------------------------------------
    def sweep(
        self,
        wanted: Optional[set[str]] = None,
        max_pages: int = 0,
        progress: Optional[Callable[[str], None]] = None,
    ) -> BulkStats:
        """Walk the entire CS2 catalogue.

        `wanted` restricts what gets WRITTEN, but every page is still paged
        through -- the endpoint has no reliable exact-name filter. If Steam
        caps the page at 10 rows this is ~3.5k requests; prefer
        `sweep_queries()` in that case.
        """
        self._paginate("", wanted, max_pages, progress)
        self.state.set("idle")
        return self.stats

    def sweep_queries(
        self,
        queries: Iterable[str],
        wanted: Optional[set[str]] = None,
        max_pages_per_query: int = 0,
        progress: Optional[Callable[[str], None]] = None,
    ) -> BulkStats:
        """Run one paginated search per query string.

        Server-side filtering is the only real defence when Steam caps the
        page size: searching "AK-47" pages through a few hundred rows instead
        of 35,000. Pass the weapon names your tier actually needs.
        """
        qs = [q for q in queries if q]
        for i, q in enumerate(qs, 1):
            self.stats.queries_done = i
            self._paginate(q, wanted, max_pages_per_query, progress,
                           label=f"[{i}/{len(qs)}] {q!r} ")
            if self.page_delay > 0 and i < len(qs):
                time.sleep(self.page_delay
                           + random.uniform(0, self.page_delay * 0.25))
        self.state.set("idle")
        return self.stats

"""Steam Community Market price client (single-item endpoint).

`/market/priceoverview/` returns ONE item per request and is aggressively
rate limited **per IP** -- a 429 ban commonly lasts hours and re-arms every
time you hit it while banned. Use `bulk.py` to fill an empty cache; keep
this for precise confirmation of a short shortlist, where `median` and 24h
`volume` actually matter.

OBSERVABILITY: every blocking wait publishes its reason to `self.state`, so
a caller can render a live heartbeat. Without it, a long backoff sleep and
a genuine hang look identical from the outside.

ENCODING: every JSON read/write pins `encoding="utf-8"`. Market hash names
contain characters cp1252 cannot represent; on Windows the default codec
raises UnicodeEncodeError. Never drop the explicit encoding.
"""
from __future__ import annotations

import json
import random
import re
import sqlite3
import threading
import time
from dataclasses import dataclass, field
from pathlib import Path
from typing import Optional, Union

import httpx

APPID_CS2 = 730

# Steam currency ids
CURRENCY = {"USD": 1, "GBP": 2, "EUR": 3, "BRL": 7, "RUB": 5, "CNY": 23}

PRICE_RE = re.compile(r"[\d]+(?:[.,]\d+)*(?:[.,]\d{2})?")


def parse_price(raw: Optional[str]) -> Optional[float]:
    """Parse Steam's localized price strings ('R$ 1.234,56', '$12.34')."""
    if not raw:
        return None
    m = PRICE_RE.search(raw.replace("\xa0", " "))
    if not m:
        return None
    s = m.group(0)
    if "," in s and "." in s:
        s = s.replace(".", "").replace(",", ".") if s.rfind(",") > s.rfind(".") else s.replace(",", "")
    elif "," in s:
        s = s.replace(",", ".") if len(s.split(",")[-1]) == 2 else s.replace(",", "")
    try:
        return float(s)
    except ValueError:
        return None


@dataclass
class ClientState:
    """What the client is doing right now, for live progress rendering."""

    phase: str = "idle"          # idle | throttling | requesting | backoff
    item: str = ""
    detail: str = ""
    since: float = field(default_factory=time.monotonic)

    def set(self, phase: str, detail: str = "") -> None:
        self.phase, self.detail, self.since = phase, detail, time.monotonic()

    @property
    def elapsed(self) -> float:
        return time.monotonic() - self.since


class TokenBucket:
    """Thread-safe token bucket: `rate` requests per `per` seconds."""

    def __init__(self, rate: int = 18, per: float = 60.0):
        self.rate, self.per = rate, per
        self._tokens = float(rate)
        self._last = time.monotonic()
        self._lock = threading.Lock()

    def take(self, state: Optional[ClientState] = None) -> None:
        while True:
            with self._lock:
                now = time.monotonic()
                self._tokens = min(
                    self.rate, self._tokens + (now - self._last) * self.rate / self.per)
                self._last = now
                if self._tokens >= 1.0:
                    self._tokens -= 1.0
                    return
                wait = (1.0 - self._tokens) * self.per / self.rate
            if state:
                state.set("throttling", f"pacing {wait:.1f}s")
            time.sleep(wait + random.uniform(0.05, 0.35))


@dataclass(slots=True)
class Quote:
    market_hash_name: str
    lowest: Optional[float]
    median: Optional[float]
    volume: Optional[int]
    fetched_at: float

    @property
    def best(self) -> Optional[float]:
        """Conservative reference price.

        For an OUTPUT you plan to sell, `median` is the honest number: the
        lowest listing is often a single outlier you cannot reproduce.
        For an INPUT you plan to buy, `lowest` is what you actually pay.
        """
        return self.median or self.lowest


class PriceCache:
    def __init__(self, path: Union[str, Path]):
        path = Path(path)
        path.parent.mkdir(parents=True, exist_ok=True)
        self.db = sqlite3.connect(str(path), check_same_thread=False)
        self.db.execute(
            """CREATE TABLE IF NOT EXISTS prices(
                 name TEXT, currency INT, lowest REAL, median REAL,
                 volume INT, ts REAL, PRIMARY KEY(name, currency))""")
        # remembers names Steam has no listing for, so they are not re-fetched
        # on every run -- these are a large share of a full sweep
        self.db.execute(
            """CREATE TABLE IF NOT EXISTS misses(
                 name TEXT, currency INT, ts REAL, PRIMARY KEY(name, currency))""")
        self.db.commit()
        self._lock = threading.Lock()

    def get(self, name: str, currency: int, ttl: float) -> Optional[Quote]:
        cur = self.db.execute(
            "SELECT lowest, median, volume, ts FROM prices WHERE name=? AND currency=?",
            (name, currency)).fetchone()
        if not cur:
            return None
        lo, med, vol, ts = cur
        if time.time() - ts > ttl:
            return None
        return Quote(name, lo, med, vol, ts)

    def is_known_miss(self, name: str, currency: int, ttl: float) -> bool:
        row = self.db.execute(
            "SELECT ts FROM misses WHERE name=? AND currency=?",
            (name, currency)).fetchone()
        return bool(row) and (time.time() - row[0] <= ttl)

    def put(self, q: Quote, currency: int) -> None:
        with self._lock:
            self.db.execute(
                "INSERT OR REPLACE INTO prices VALUES (?,?,?,?,?,?)",
                (q.market_hash_name, currency, q.lowest, q.median, q.volume,
                 q.fetched_at))
            self.db.commit()

    def put_miss(self, name: str, currency: int) -> None:
        with self._lock:
            self.db.execute("INSERT OR REPLACE INTO misses VALUES (?,?,?)",
                            (name, currency, time.time()))
            self.db.commit()

    def export(self, path: Union[str, Path], currency: int) -> int:
        rows = self.db.execute(
            "SELECT name, lowest, median, volume FROM prices WHERE currency=?",
            (currency,)).fetchall()
        payload = {n: {"lowest": lo, "median": me, "volume": v}
                   for n, lo, me, v in rows}
        Path(path).write_text(json.dumps(payload, ensure_ascii=False),
                              encoding="utf-8")
        return len(payload)

            return cached
        if self.cache.is_known_miss(market_hash_name, self.currency, self.miss_ttl):
            self.stats["empty"] += 1
            self.state.set("idle")
            return None
        self.stats["misses"] += 1

        backoff = min(5.0, self.max_backoff)
        for attempt in range(self.max_retries):
            self.bucket.take(self.state)
            self.state.set("requesting", f"attempt {attempt + 1}")
            try:
                r = self.client.get(self.URL, params={
                    "appid": APPID_CS2,
                    "currency": self.currency,
                    "market_hash_name": market_hash_name,
                })
            except httpx.TimeoutException:
                self.stats["timeouts"] += 1
                self.state.set("backoff", f"timeout, waiting {backoff:.0f}s")
                time.sleep(backoff)
                backoff = min(backoff * 2, self.max_backoff)
                continue
            except httpx.HTTPError as e:
                self.stats["errors"] += 1
                self.state.set("backoff", f"{type(e).__name__}, waiting {backoff:.0f}s")
                time.sleep(backoff)
                backoff = min(backoff * 2, self.max_backoff)
                continue

            if r.status_code == 429:
                self.stats["429s"] += 1
                ra = float(r.headers.get("Retry-After", 0) or 0)
                wait = min(max(ra, backoff), self.max_backoff)
                self.state.set("backoff", f"429, waiting {wait:.0f}s")
                time.sleep(wait + random.uniform(0, 2))
                backoff = min(backoff * 2, self.max_backoff)
                continue
            if r.status_code == 500:
                # Steam returns 500 for names that do not exist on the market
                self.cache.put_miss(market_hash_name, self.currency)
                self.state.set("idle")
                return None
            if r.status_code != 200:
                self.stats["errors"] += 1
                self.state.set("backoff", f"HTTP {r.status_code}, waiting {backoff:.0f}s")
                time.sleep(backoff)
                backoff = min(backoff * 2, self.max_backoff)
                continue

            try:
                data = r.json()
            except ValueError:
                # Steam serves an HTML maintenance/error page with a 200
                self.stats["errors"] += 1
                self.state.set("backoff", f"non-JSON body, waiting {backoff:.0f}s")
                time.sleep(backoff)
                backoff = min(backoff * 2, self.max_backoff)
                continue

            if not data.get("success"):
                self.cache.put_miss(market_hash_name, self.currency)
                self.state.set("idle")
                return None
            q = Quote(
                market_hash_name,
                parse_price(data.get("lowest_price")),
                parse_price(data.get("median_price")),
                int(str(data.get("volume", "0")).replace(",", "") or 0),
                time.time(),
            )
            self.cache.put(q, self.currency)
            self.state.set("idle")
            return q

        # Out of retries: record it so a resumed run does not stall here again
        self.stats["gave_up"] += 1
        self.state.set("idle")
        return None


class PriceBook:
    """Offline price source: a plain {market_hash_name: {...}} JSON snapshot."""

    def __init__(self, data: dict):
        self.data = data

    @classmethod
    def load(cls, path: Union[str, Path]) -> "PriceBook":
        return cls(json.loads(Path(path).read_text(encoding="utf-8")))

    def get(self, market_hash_name: str) -> Optional[Quote]:
        d = self.data.get(market_hash_name)
        if d is None:
            return None
        if isinstance(d, (int, float)):
            return Quote(market_hash_name, float(d), float(d), None, time.time())
        return Quote(market_hash_name, d.get("lowest"), d.get("median"),
                     d.get("volume"), time.time())

@app.post("/api/evaluate")
def evaluate(req: EvaluateRequest):
    db = get_db()
    prices = get_prices(req.offline, req.currency, 0, 0, 0.0)
    items = []
    for spec in req.inputs:
        skin = (db.by_id(f"{spec.name}|{spec.collection}") if spec.collection
                else db.by_name(spec.name))
        if skin is None:
            raise HTTPException(404, f"Unknown skin: {spec.name}")
        items.append(InputItem(skin, spec.float_value, spec.price,
                               req.stattrak, spec.souvenir))
    if not items:
        raise HTTPException(400, "No inputs")
    res = evaluate_contract(items, db.pool, prices.price_fn(),
                            odds_model=req.odds_model, fee_rate=req.fee_rate,
                            stattrak=req.stattrak, skin_by_id=db.by_id)
    fake = type("C", (), {"result": res, "target_collection": items[0].skin.collection,
                          "filler_collection": None, "k_target": len(items)})()
    payload = explain(fake)
    payload["rule_warnings"] = validate_contract(items)
    return payload


@app.post("/api/scan")
def scan(req: ScanRequest):
    db = get_db()
    prices = get_prices(req.offline, req.currency, req.min_volume,
                        req.min_input_volume, req.haircut)
    r = RARITIES.get(req.rarity.lower())
    if r is None:
        raise HTTPException(400, f"Bad rarity: {req.rarity}")
    cands = search(db, prices, r, budget=req.budget, odds_model=req.odds_model,
                   fee_rate=req.fee_rate, stattrak=req.stattrak,
                   souvenir=req.souvenir, float_percentile=req.float_percentile,
                   min_roi=req.min_roi, top_n=req.top * 3)
    if req.robust:
        cands = robust_filter(cands, db, prices, fee_rate=req.fee_rate)
    return {"count": len(cands[:req.top]),
            "results": [explain(c) for c in cands[:req.top]]}

def _safe_console() -> Console:
    """A Console that never dies on a legacy Windows code page."""
    enc = (getattr(sys.stdout, "encoding", "") or "").lower().replace("-", "")
    if enc not in ("utf8", "utf8mb4"):
        try:  # Python 3.7+: retrofit the stream in place
            sys.stdout.reconfigure(encoding="utf-8", errors="replace")
            sys.stderr.reconfigure(encoding="utf-8", errors="replace")
        except Exception:
            return Console(legacy_windows=True, safe_box=True)
    return Console()

def _tier_names(db: SkinDB, r: Rarity, souvenir: bool, include_outputs: bool) -> set[str]:
    """Every market_hash_name a scan at `r` could need."""
    names: set[str] = set()
    for s in (x for c in db.collections(r) for x in db.pool(c, r)):
        for w in s.reachable_wears():
            names.add(s.market_hash_name(w))
            if souvenir and s.souvenir_capable:
                names.add(s.market_hash_name(w, souvenir=True))
    nxt = r.next_tier()
    if include_outputs and nxt:  # outputs are always normal quality
        for s in (x for c in db.collections(nxt) for x in db.pool(c, nxt)):
            for w in s.reachable_wears():
                names.add(s.market_hash_name(w))
    return names

@app.command("cooldown")
def cooldown(
    currency: str = "USD",
    wait: bool = typer.Option(False, help="Poll until the ban clears"),
    poll: float = typer.Option(300.0, help="Seconds between polls when waiting"),
):
    """Check whether this IP is currently rate limited by Steam.

    The limit is per IP, not per account, and a ban can last many hours. It
    also re-arms every time you hit it while banned, so hammering it keeps
    you locked out. If this reports a ban, stop all requests and wait.
    """
    market = SteamMarket(currency=currency, rate=60, max_retries=1)
    probe = "AK-47 | Elite Build (Field-Tested)"
    try:
        while True:
            t0 = time.monotonic()
            try:
                r = market.client.get(market.URL, params={
                    "appid": 730, "currency": market.currency,
                    "market_hash_name": probe})
                dt = time.monotonic() - t0
                if r.status_code == 200 and '"success":true' in r.text.lower():
                    console.print(f"[green]OK -- not rate limited ({dt:.1f}s).[/green]")
                    raise typer.Exit(0)
                if r.status_code == 429:
                    ra = r.headers.get("Retry-After")
                    console.print(f"[red]429 -- this IP is rate limited"
                                  f"{f' (Retry-After {ra}s)' if ra else ''}.[/red]")
                    console.print("[yellow]Stop all requests. Every attempt while "
                                  "banned can extend it. Bans commonly last hours; "
                                  "a router reboot that changes your IP also clears "
                                  "it.[/yellow]")
                else:
                    console.print(f"[yellow]HTTP {r.status_code} ({dt:.1f}s) -- "
                                  "not a clean success.[/yellow]")
            except typer.Exit:
                raise
            except Exception as e:
                console.print(f"[red]{type(e).__name__}: {e}[/red]")
            if not wait:
                raise typer.Exit(1)
            console.print(f"[dim]sleeping {poll:.0f}s...[/dim]")
            time.sleep(poll)
    finally:
        market.close()

@app.command("bulk-warm")
def bulk_warm(
    currency: str = "USD",
    rarity: Optional[str] = typer.Option(
        None, help="Only keep names for this tier (default: cache everything)"),
    souvenir: bool = typer.Option(True, help="Include Souvenir variants when filtering"),
    page_delay: float = typer.Option(6.0, help="Seconds between pages. Raise if 429s."),
    max_pages: int = typer.Option(0, help="Stop after N pages (0 = all)"),
    export: bool = typer.Option(True, help="Write data/price_snapshot.json at the end"),
):
    """Price the whole market in ~60 requests instead of ~5600.

    Uses Steam's paginated market search (100 items per request) -- the same
    endpoint the market web page calls -- rather than one-item-at-a-time
    priceoverview. This is the recommended way to fill an empty cache.

    Caveat: bulk data has the lowest listing price and a LISTING COUNT, but
    no true median and no 24h sales volume. Run `warm-prices` afterwards on
    the handful of names in your shortlist to get precise figures.
    """
    wanted: Optional[set[str]] = None
    if rarity:
        db = SkinDB.load()
        r = RARITIES[rarity.lower()]
        wanted = _tier_names(db, r, souvenir, include_outputs=True)
        console.print(f"Filtering to {len(wanted)} names for {r.label} "
                      "(plus its output tier).")

    market = BulkMarket(currency=currency, page_delay=page_delay)
    console.print("[dim]Walking the CS2 market 100 items per request. "
                  "Ctrl+C is safe - every page is committed as it arrives.[/dim]")
    t0 = time.monotonic()
    try:
        market.sweep(wanted=wanted, max_pages=max_pages,
                     progress=lambda m: console.print(f"[dim]{m}[/dim]"))
    except KeyboardInterrupt:
        console.print("\n[yellow]Interrupted -- pages fetched so far are cached.[/yellow]")
    finally:
        market.close()

    s = market.stats
    t = Table(title=f"bulk-warm summary ({(time.monotonic()-t0)/60:.1f} min)")
    t.add_column("Metric")
    t.add_column("Count", justify="right")
    for label, v in (("pages fetched", s.pages), ("items seen", s.items),
                     ("quotes cached", s.written), ("rate limited", s.rate_limited),
                     ("retries", s.retries), ("errors", s.errors)):
        t.add_row(label, str(v))
    console.print(t)
    if s.rate_limited:
        console.print(f"[yellow]Hit {s.rate_limited} rate limits. Re-run with "
                      f"--page-delay {page_delay * 2:.0f}.[/yellow]")
    if export:
        n = market.cache.export(Path("data/price_snapshot.json"), market.currency)
        console.print(f"[green]{n} quotes -> data/price_snapshot.json[/green]")
    console.print("[dim]Bulk quotes have no median/volume. Run 'warm-prices' on "
                  "your shortlist before committing money.[/dim]")

def _heartbeat(market, total, counter, stop) -> None:
    """Render live progress from the client's own state, once a second."""
    with console.status("") as st:
        while not stop.is_set():
            s = market.state
            phase = s.phase
            if phase in ("backoff", "throttling") and s.elapsed > 2:
                phase = f"[yellow]{phase} {s.elapsed:.0f}s[/yellow]"
            elif phase == "requesting" and s.elapsed > 10:
                phase = f"[yellow]requesting {s.elapsed:.0f}s[/yellow]"
            st.update(
                f"[{counter['i']}/{total}] {phase} {s.detail} | "
                f"cached={market.stats['hits'] + market.stats['empty']} "
                f"new={market.stats['misses']} 429={market.stats['429s']} "
                f"timeout={market.stats['timeouts']} err={market.stats['errors']} "
                f"| {s.item[:44]}")
            stop.wait(1.0)


@app.command("warm-prices")
def warm_prices(
    rarity: str = typer.Option("restricted"),
    currency: str = "USD",
    ttl: float = 8.0,
    include_outputs: bool = True,
    souvenir: bool = typer.Option(True, help="Also fetch Souvenir input variants"),
    rate: int = typer.Option(18, help="Requests per minute. Lower it if you see 429s."),
    timeout: float = typer.Option(20.0, help="Per-request timeout in seconds"),
    limit: int = typer.Option(0, help="Stop after N names (0 = all). Good for a test run."),
    verbose: bool = typer.Option(False, help="Log every network fetch as it happens"),
):
    """Precise per-item prices (median + 24h volume) for one tier.

    SLOW: one request per name. Prefer `bulk-warm` to fill an empty cache,
    and use this only to confirm a shortlist before committing money.
    """
    db = SkinDB.load()
    r = RARITIES[rarity.lower()]
    market = SteamMarket(currency=currency, ttl_hours=ttl, rate=rate, timeout=timeout)

    ordered = sorted(_tier_names(db, r, souvenir, include_outputs))
    if limit:
        ordered = ordered[:limit]

    # Count what is genuinely missing so the ETA reflects real work
    todo = [n for n in ordered
            if market.cache.get(n, market.currency, market.ttl) is None
            and not market.cache.is_known_miss(n, market.currency, market.miss_ttl)]
    console.print(f"{len(ordered)} names; {len(ordered) - len(todo)} already cached, "
                  f"{len(todo)} to fetch at {rate}/min "
                  f"(~{len(todo)/max(rate,1):.0f} min). "
                  "Ctrl+C is safe: progress is cached and resumes.")
    if not todo:
        console.print("[green]Nothing to fetch -- cache is already complete.[/green]")

    counter = {"i": 0}
    stop = threading.Event()
    hb = None
    if not verbose:
        hb = threading.Thread(target=_heartbeat,
                              args=(market, len(ordered), counter, stop), daemon=True)
        hb.start()

    t0 = time.monotonic()
    try:
        for i, n in enumerate(ordered, 1):
            counter["i"] = i
            fresh = False
            s0 = t0
            if verbose:
                fresh = (market.cache.get(n, market.currency, market.ttl) is None
                         and not market.cache.is_known_miss(
                             n, market.currency, market.miss_ttl))
                if fresh:
                    console.print(f"[dim][{i}/{len(ordered)}] fetching {n}[/dim]")
                    s0 = time.monotonic()
            market.get(n)
            if verbose and fresh:
                console.print(f"[dim]    done in {time.monotonic()-s0:.1f}s[/dim]")
    except KeyboardInterrupt:
        console.print("\n[yellow]Interrupted -- everything fetched so far is "
                      "cached. Re-run to resume.[/yellow]")
    finally:
        stop.set()
        if hb:
            hb.join(timeout=2)

    mins = (time.monotonic() - t0) / 60
    n_export = market.cache.export(Path("data/price_snapshot.json"), market.currency)
    market.close()

    t = Table(title=f"warm-prices summary ({mins:.1f} min)")
    t.add_column("Metric")
    t.add_column("Count", justify="right")
    for k, label in (("hits", "from cache"), ("misses", "fetched"),
                     ("empty", "no listing"), ("429s", "rate limited"),
                     ("timeouts", "timed out"), ("errors", "errors"),
                     ("gave_up", "gave up after retries")):
        t.add_row(label, str(market.stats[k]))
    console.print(t)
    console.print(f"[green]{n_export} quotes -> data/price_snapshot.json[/green]")
    if market.stats["429s"] > max(market.stats["misses"], 1) * 0.05:
        console.print(f"[yellow]Over 5% of fetches were rate limited. Re-run with "
                      f"--rate {max(6, rate // 2)}, or use bulk-warm.[/yellow]")

@app.command("verify-souvenir")
def verify_souvenir(
    rarity: str = typer.Option("consumer"),
    currency: str = "USD",
    offline: Optional[Path] = typer.Option(None, help="Use a price snapshot JSON"),
    sample: int = typer.Option(40, help="How many skins to probe"),
    rate: int = typer.Option(10),
):
    """Check whether 'Souvenir <skin>' listings actually exist on the market.

    The skin database can only infer which finishes have a Souvenir variant.
    This asks the market directly for a sample and reports the hit rate.
    """
    db = SkinDB.load()
    r = RARITIES[rarity.lower()]
    policy = PricingPolicy(min_input_volume=1)
    prices = (PriceService(PriceBook.load(offline), policy) if offline
              else PriceService(SteamMarket(currency=currency, rate=rate), policy))

    cands = [s for c in db.collections(r) for s in db.pool(c, r) if s.souvenir_capable]
    if not cands:
        console.print(f"[yellow]No {r.label} skin is marked Souvenir-capable.[/yellow]")
        raise typer.Exit(0)

    probe = cands[:: max(1, len(cands) // max(sample, 1))][:sample]
    console.print(f"Probing {len(probe)} of {len(cands)} Souvenir-capable "
                  f"{r.label} skins...")

    hit = miss = 0
    t = Table(title="Souvenir listing check")
    for c in ("Skin", "Wear", "Souvenir price", "Normal price"):
        t.add_column(c, justify="left" if c == "Skin" else "right")
    with console.status("") as st:
        for i, s in enumerate(probe, 1):
            wears = s.reachable_wears()
            if not wears:
                continue
            wear = wears[len(wears) // 2]
            sv = prices.buy_price(s, wear, souvenir=True)
            nm = prices.buy_price(s, wear, souvenir=False)
            if sv:
                hit += 1
                if hit <= 15:
                    t.add_row(s.name[:38], wear, f"{sv:.2f}",
                              f"{nm:.2f}" if nm else "-")
            else:
                miss += 1
            st.update(f"[{i}/{len(probe)}] found={hit} missing={miss}")
    if hit:
        console.print(t)
    pct = hit / max(hit + miss, 1)
    console.print(f"[{'green' if pct > 0.5 else 'yellow'}]"
                  f"{hit}/{hit+miss} ({pct:.0%}) had a real Souvenir listing.[/]")
    if pct < 0.5:
        console.print("[yellow]The Souvenir flag is over-reporting. Scans stay "
                      "correct (unpriced units are skipped) but warm-prices "
                      "wastes requests. Consider --no-souvenir.[/yellow]")

@app.command("eval")
def eval_recipe(
    recipe: Path = typer.Argument(
        ..., help="JSON: [{name, collection, float, price, souvenir}, ...]"),
    offline: Optional[Path] = None,
    currency: str = "USD",
    odds: str = "outcome_weighted",
    fee: Optional[float] = None,
    stattrak: bool = False,
):
    """Score a hand-built contract."""
    db = SkinDB.load()
    prices = _price_service(offline, currency, 8.0, PricingPolicy())
    items = json.loads(Path(recipe).read_text(encoding="utf-8"))
    inputs = []
    for it in items:
        sid = f"{it['name']}|{it.get('collection')}"
        skin = db.by_id(sid) or db.by_name(it["name"])
        if skin is None:
            raise typer.BadParameter(f"Unknown skin: {sid}")
        inputs.append(InputItem(skin, float(it["float"]), float(it.get("price", 0)),
                                stattrak, bool(it.get("souvenir", False))))
    for p in validate_contract(inputs):
        console.print(f"[yellow]Rule warning: {p}[/yellow]")
    need = contract_size(inputs[0].skin.rarity)
    if len(inputs) != need:
        console.print(f"[yellow]Warning: {len(inputs)} inputs, "
                      f"contract expects {need}[/yellow]")
    res = evaluate_contract(inputs, db.pool, prices.price_fn(), odds_model=odds,
                            fee_rate=fee, stattrak=stattrak, skin_by_id=db.by_id)
    payload = explain(type("C", (), {
        "result": res, "target_collection": inputs[0].skin.collection,
        "filler_collection": None, "k_target": len(inputs)})())
    console.print_json(json.dumps(payload, indent=2, ensure_ascii=False))

def _weapon_terms(db, r, souvenir: bool) -> list[str]:
    """Distinct weapon names for a tier, usable as server-side search terms.

    'AK-47 | Elite Build' -> 'AK-47'. Searching per weapon pages through a
    few hundred rows instead of the whole 35k catalogue, which is what makes
    this viable when Steam caps the page size.

        for c in db.collections(tier):
            for s in db.pool(c, tier):
                head = s.name.split("|")[0].strip()
                if head:
                    terms.add(head)
    return sorted(terms)

"""Trade-up math: float mapping, outcome odds, expected value.

Terminology follows the in-game model (post-Retakes Update):

  Float Value          the raw wear rating on the item, 0.00-1.00
  Float Cap            a skin's own [min_float, max_float]; not every skin
                       spans the full range
  Trade-up Float       the float rescaled onto a universal 0-1 axis using the
                       skin's OWN cap:  (f - min) / (max - min)
                       Also called the Adjusted Float.
  Average Trade-up     the mean of the inputs' Trade-up Floats. NOT the mean
  Float                of the raw floats -- that is the pre-Retakes model and
                       gives wrong answers whenever caps differ.
  Outcome Float        the Average Trade-up Float rescaled back into the
                       OUTPUT skin's cap: avg * (omax - omin) + omin

Verified against known-good worked examples to 8 decimal places:

  P90 | Trigon 0.749 in cap 0.08-0.75      -> trade-up float 0.99850746
  10 of them                               -> average       0.99850746
  onto AWP | Asiimov cap 0.18-1.00         -> outcome float 0.99877612
  onto AUG | Chameleon cap 0.00-0.50       -> outcome float 0.49925373
  MAC-10 | Sakkaku 0.22798392 in 0.21-0.79 -> trade-up float 0.03100676

Everything here is pure and deterministic so it can be unit-tested without
network access. Prices are injected via a `price_fn` callback.
"""
from __future__ import annotations

import math
from collections import Counter
from typing import Callable, Literal, Optional, Sequence

from .models import ContractResult, InputItem, Outcome, Rarity, Skin, wear_from_float

# "per_input" is the CURRENT in-game rule (post-Retakes Update).
# "outcome_weighted" is the LEGACY pre-Retakes rule, kept only so you can
# diff against older calculators. Do not price real contracts with it.
OddsModel = Literal["per_input", "outcome_weighted"]

DEFAULT_ODDS_MODEL: OddsModel = "per_input"

STEAM_PLATFORM_FEE = 0.05   # Valve
STEAM_GAME_FEE = 0.10       # CS2 publisher fee
STEAM_TOTAL_FEE = STEAM_PLATFORM_FEE + STEAM_GAME_FEE


# ---------------------------------------------------------------------------
# Fees
# ---------------------------------------------------------------------------
def steam_net_from_buyer_price(buyer_price: float) -> float:
    """Seller proceeds when the BUYER pays `buyer_price`.

    Steam builds the fee ON TOP of the seller's cut:
        buyer = seller * (1 + 0.05 + 0.10)
    so the seller keeps buyer / 1.15 (~86.96%), not buyer * 0.85.
    A $0.01 minimum applies to each of the two components.
    """
    if buyer_price <= 0:
        return 0.0
    seller = buyer_price / (1.0 + STEAM_TOTAL_FEE)
    platform = max(0.01, math.floor(seller * STEAM_PLATFORM_FEE * 100) / 100)
    game = max(0.01, math.floor(seller * STEAM_GAME_FEE * 100) / 100)
    return round(max(0.0, buyer_price - platform - game), 4)


def net_after_fee(buyer_price: float, fee_rate: Optional[float] = None) -> float:
    """Generic marketplace net. `fee_rate=None` -> exact Steam model."""
    if fee_rate is None:
        return steam_net_from_buyer_price(buyer_price)
    return round(buyer_price * (1.0 - fee_rate), 4)


# ---------------------------------------------------------------------------
# Float mapping
# ---------------------------------------------------------------------------
def trade_up_float(float_value: float, min_float: float, max_float: float) -> float:
    """Rescale a raw float onto the universal 0-1 axis using the skin's cap.

    Also known as the Adjusted Float. This is why two items showing the same
    raw float can contribute very differently: 0.10 on a 0.00-1.00 skin is
    0.10, but 0.10 on a 0.00-0.50 skin is 0.20 -- already 20% through its
    own range.
    """
    span = max_float - min_float
    if span <= 0:
        return 0.0
    return (float_value - min_float) / span


def average_trade_up_float(inputs: Sequence[InputItem]) -> float:
    """Mean of the inputs' Trade-up Floats.

    NOT the mean of raw floats. Averaging raw floats is the pre-Retakes
    model and silently misprices any contract whose inputs have different
    float caps.
    """
    if not inputs:
        return 0.0
    return sum(i.trade_up_float for i in inputs) / len(inputs)


def outcome_float(avg_trade_up_float: float, out_skin: Skin) -> float:
    """Rescale the Average Trade-up Float into the OUTPUT skin's own cap.

    Each possible outcome has its own cap, so one contract yields a
    different outcome float -- and possibly a different wear tier -- per
    outcome skin.
    """
    return (avg_trade_up_float * (out_skin.max_float - out_skin.min_float)
            + out_skin.min_float)


# Backwards-compatible aliases (older code called these normalized/average)
normalized_float = trade_up_float
average_normalized = average_trade_up_float


def required_avg_for_wear(out_skin: Skin, target_wear: str) -> tuple[float, float]:
    """The (lo, hi) Average Trade-up Float window that lands `out_skin` in
    `target_wear`. Returns (nan, nan) if that wear is unreachable."""
    from .models import WEAR_BRACKETS

    bracket = next((b for b in WEAR_BRACKETS if b[0] == target_wear), None)
    if bracket is None:
        return (float("nan"), float("nan"))
    _, w_lo, w_hi = bracket
    span = out_skin.max_float - out_skin.min_float
    if span <= 0:
        return ((0.0, 1.0) if w_lo <= out_skin.min_float < w_hi
                else (float("nan"), float("nan")))
    lo = (max(w_lo, out_skin.min_float) - out_skin.min_float) / span
    hi = (min(w_hi, out_skin.max_float) - out_skin.min_float) / span
    if hi <= lo:
        return (float("nan"), float("nan"))
    return (max(0.0, lo), min(1.0, hi))


required_avg_norm_for_wear = required_avg_for_wear  # legacy alias

def validate_contract(inputs: Sequence[InputItem]) -> list[str]:
    """Rule violations that would make the in-game contract impossible."""
    problems: list[str] = []
    if not inputs:
        return ["no inputs"]
    rarities = {i.skin.rarity for i in inputs}
    if len(rarities) > 1:
        problems.append("inputs span multiple rarities: "
                        + ", ".join(sorted(r.label for r in rarities)))
    has_st = any(i.stattrak for i in inputs)
    has_sv = any(i.souvenir for i in inputs)
    if has_st and not all(i.stattrak for i in inputs):
        problems.append("StatTrak must be all-or-nothing")
    if has_st and has_sv:
        problems.append("StatTrak items cannot be mixed with Souvenir items")
    for i in inputs:
        if not (i.skin.min_float <= i.float_value <= i.skin.max_float):
            problems.append(
                f"{i.skin.name} float {i.float_value:.4f} is outside its cap "
                f"{i.skin.min_float:.2f}-{i.skin.max_float:.2f}")
        if i.souvenir and not i.skin.souvenir_capable:
            problems.append(f"{i.skin.name} has no Souvenir variant")
        if i.stattrak and not i.skin.stattrak_capable:
            problems.append(f"{i.skin.name} has no StatTrak variant")
    return problems

# Outcome probabilities
# ---------------------------------------------------------------------------
def collection_odds(
    inputs: Sequence[InputItem],
    pool: Callable[[str, Rarity], list[Skin]],
) -> dict[str, float]:
    """P(the outcome comes from each collection), current rules.

    Each input contributes an equal 1/n share, so a collection's chance is
    simply how many of your inputs came from it:

        P(collection C) = inputs_from_C / n

    Seven inputs from one collection is a 70% chance of hitting that
    collection, regardless of how many outcomes any collection has.
    """
    if not inputs:
        return {}
    tier = inputs[0].skin.rarity.next_tier()
    if tier is None:
        return {}
    counts = Counter(i.skin.collection for i in inputs)
    # collections with no higher tier are dead ends: they contribute no
    # outcome mass, and the remaining share is renormalized over the rest
    live = {c: n for c, n in counts.items() if pool(c, tier)}
    total = sum(live.values())
    if not total:
        return {}
    return {c: n / total for c, n in live.items()}


def outcome_probabilities(
    inputs: Sequence[InputItem],
    pool: Callable[[str, Rarity], list[Skin]],
    model: OddsModel = DEFAULT_ODDS_MODEL,
) -> dict[str, float]:
    """Probability per output skin id.

    `per_input` -- CURRENT in-game rule. Odds are per item, not per
    collection outcome count. Each input carries 1/n; that share is then
    split evenly across its collection's outcomes at the next tier:

        P(skin s in collection C) = (inputs_from_C / n) / k_C

    where k_C is the number of outcomes collection C offers. So one
    M4A1-S | Knight among ten inputs is a flat 10% at Dragon Lore, because
    Cobblestone offers exactly one outcome above it -- while a collection
    offering two outcomes yields 5% each for the same single input.

    `outcome_weighted` -- LEGACY pre-Retakes rule, where a collection with
    more outcomes absorbed proportionally more total mass:

        P(skin s in C) = inputs_from_C / sum_k(inputs_from_k * k_k)

    On the ten-input example above this returns a uniform 5.26% for every
    outcome, which no longer matches the game. Kept only for comparison
    against older calculators.
    """
    if not inputs:
        return {}
    tier = inputs[0].skin.rarity.next_tier()
    if tier is None:
        return {}

    counts = Counter(i.skin.collection for i in inputs)
    pools = {c: pool(c, tier) for c in counts}
    pools = {c: p for c, p in pools.items() if p}
    if not pools:
        return {}

    probs: dict[str, float] = {}
    if model == "per_input":
        n = sum(counts[c] for c in pools)
        for c, skins in pools.items():
            share = counts[c] / n          # collection odds
            each = share / len(skins)      # split evenly across its outcomes
            for s in skins:
                probs[s.id] = probs.get(s.id, 0.0) + each
    elif model == "outcome_weighted":
        denom = sum(counts[c] * len(pools[c]) for c in pools)
        for c, skins in pools.items():
            for s in skins:
                probs[s.id] = probs.get(s.id, 0.0) + counts[c] / denom
    else:
        raise ValueError(f"unknown odds model: {model!r}")

    total = sum(probs.values())
    return {k: v / total for k, v in probs.items()}


# ---------------------------------------------------------------------------
# Full contract evaluation
# ---------------------------------------------------------------------------
def evaluate_contract(
    inputs: Sequence[InputItem],
    pool: Callable[[str, Rarity], list[Skin]],
    price_fn: Callable[[Skin, str, bool], Optional[float]],
    *,
    odds_model: OddsModel = DEFAULT_ODDS_MODEL,
    fee_rate: Optional[float] = None,
    stattrak: bool = False,
    skin_by_id: Optional[Callable[[str], Skin]] = None,
) -> ContractResult:
    """Evaluate one concrete contract.

    `price_fn(skin, wear, stattrak)` prices the OUTPUT. It is never called
    with a souvenir flag: a trade-up output is always normal quality, even
    when every input was a Souvenir.
    """
    inputs = list(inputs)
    res = ContractResult(inputs=inputs, stattrak=stattrak)
    if not inputs:
        return res

    res.avg_normalized = average_trade_up_float(inputs)
    res.cost = round(sum(i.price for i in inputs), 4)
    res.collection_odds = collection_odds(inputs, pool)

    probs = outcome_probabilities(inputs, pool, odds_model)
    tier = inputs[0].skin.rarity.next_tier()
    lookup: dict[str, Skin] = {}
    for c in {i.skin.collection for i in inputs}:
        for s in pool(c, tier):
            lookup[s.id] = s

    outcomes: list[Outcome] = []
    for sid, p in probs.items():
        s = lookup.get(sid) or (skin_by_id(sid) if skin_by_id else None)
        if s is None:
            continue
        f = outcome_float(res.avg_normalized, s)
        wear = wear_from_float(f)
        gross = price_fn(s, wear, stattrak)
        priced = gross is not None
        gross = gross or 0.0
        outcomes.append(Outcome(
            skin=s, probability=p, out_float=round(f, 8), wear=wear,
            gross_price=round(gross, 4), net_price=net_after_fee(gross, fee_rate),
            priced=priced))

    outcomes.sort(key=lambda o: o.net_price, reverse=True)
    res.outcomes = outcomes
    res.unpriced_mass = round(sum(o.probability for o in outcomes if not o.priced), 6)
    res.ev_net = round(sum(o.probability * o.net_price for o in outcomes), 4)
    res.profit = round(res.ev_net - res.cost, 4)
    res.roi = round(res.ev_net / res.cost, 4) if res.cost > 0 else 0.0
    res.profit_chance = round(
        sum(o.probability for o in outcomes if o.net_price > res.cost), 6)
    var = sum(o.probability * (o.net_price - res.ev_net) ** 2 for o in outcomes)
    res.stddev = round(math.sqrt(max(var, 0.0)), 4)
    return res


def kelly_fraction(res: ContractResult) -> float:
    """Rough Kelly-style sizing signal (Gaussian approximation)."""
    if res.stddev <= 0 or res.cost <= 0:
        return 0.0
    return max(0.0, min(1.0, res.profit / (res.stddev ** 2)))


def n_contracts_for_confidence(res: ContractResult, z: float = 2.0) -> Optional[int]:
    """How many identical contracts before the edge dominates the noise.

    n such that n*profit >= z * stddev * sqrt(n)  ->  n >= (z*sd/profit)^2
    """
    if res.profit <= 0 or res.stddev <= 0:
        return None
    return math.ceil((z * res.stddev / res.profit) ** 2)

"""Skin images, from the same free source as the skin database.

`skins.json` from CSGO-API already carries an `image` field per finish, so
no extra API and no key is needed -- the images are a byproduct of a sync
you already run.

    python -m tradeup.cli sync-images

writes `data/images.json` as {skin name: image url}. The web app loads it
lazily and degrades to a placeholder when a name is missing, so nothing
breaks if you never run it.

Images are hotlinked from Steam's CDN (community.cloudflare.steamstatic.com
or community.akamai.steamstatic.com). They are served to your browser
directly, so this costs your server nothing and adds no API calls.
"""
from __future__ import annotations

import json
from pathlib import Path
from typing import Optional, Union

ROOT = Path(__file__).resolve().parents[1]
IMAGES_PATH = ROOT / "data" / "images.json"

# 1x1 transparent gif -- used when a skin has no image, so the layout does
# not jump and no broken-image icon appears
PLACEHOLDER = (
    "data:image/gif;base64,R0lGODlhAQABAIAAAAAAAP///yH5BAEAAAAALAAAAAABAAEAAAIBRAA7"
)


class ImageIndex:
    """name -> image URL, with graceful fallback."""

    def __init__(self, mapping: Optional[dict[str, str]] = None):
        self.map: dict[str, str] = mapping or {}
        # index without the "| " so lookups are forgiving
        self._loose = {self._key(k): v for k, v in self.map.items()}

    @staticmethod
    def _key(name: str) -> str:
        return name.lower().replace("\u2122", "").replace("★", "").strip()

    @classmethod
    def load(cls, path: Union[str, Path] = IMAGES_PATH) -> "ImageIndex":
        p = Path(path)
        if not p.exists():
            return cls({})
        try:
            return cls(json.loads(p.read_text(encoding="utf-8")))
        except (ValueError, OSError):
            return cls({})

    def get(self, skin_name: str) -> str:
        """Image URL for a finish name like 'AK-47 | Redline'."""
        if skin_name in self.map:
            return self.map[skin_name]
        v = self._loose.get(self._key(skin_name))
        return v or PLACEHOLDER

    def __len__(self) -> int:
        return len(self.map)


def build_from_payload(raw: list) -> dict[str, str]:
    """Extract {name: image} from a CSGO-API skins.json payload."""
    out: dict[str, str] = {}
    for item in raw:
        if not isinstance(item, dict):
            continue
        name = item.get("name") or ""
        # strip a trailing "(Wear)" if the entry carries one
        if name.endswith(")") and "(" in name:
            name = name[: name.rfind("(")].strip()
        img = item.get("image")
        if name and img and name not in out:
            out[name] = img
    return out


def save(mapping: dict[str, str], path: Union[str, Path] = IMAGES_PATH) -> Path:
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(json.dumps(mapping, ensure_ascii=False, indent=0),
                 encoding="utf-8")
    return p

"""Liquidity modelling -- the difference between a listed price and a sale.

A trade-up EV built on `lowest_price` assumes you can sell instantly at the
current cheapest listing. For a popular Field-Tested rifle that is roughly
true. For an unpopular skin in Well-Worn it is fiction: there may be three
listings, no buyer has touched it in a week, and the "price" is whatever one
optimistic seller typed in.

Two distinct problems, handled separately:

1. REALIZATION -- what fraction of the listed price you actually get.
   To sell into a thin book you must undercut the current lowest, and with
   almost no sales history there is no evidence anyone pays that number.

2. CONCENTRATION -- how much of the EV rides on a single outcome.
   A contract where 99% of expected value comes from one illiquid skin is
   not an edge with variance; it is a bet that one questionable price is
   real. `n*` does not capture this, because `n*` assumes the prices are
   correct and only the roll is random.

Both are reported explicitly rather than silently folded into one number,
so you can see WHY a recipe was downgraded.
"""
from __future__ import annotations

from dataclasses import dataclass
from enum import IntEnum
from typing import Optional, Sequence


class LiquidityTier(IntEnum):
    DEAD = 0        # essentially unsellable at the listed price
    THIN = 1        # sellable, but expect to undercut and wait
    OK = 2          # normal retail depth
    DEEP = 3        # sell instantly at market

    @property
    def label(self) -> str:
        return {0: "dead", 1: "thin", 2: "ok", 3: "deep"}[int(self)]


@dataclass(frozen=True, slots=True)
class LiquidityPolicy:
    """How listing depth and sales volume translate into a haircut.

    `listings` is SELL LISTINGS (supply) -- what bulk-warm gives you.
    `volume` is 24h SALES -- what warm-prices gives you, and far stronger
    evidence. When both are present, volume wins.

    Realization factors are deliberately blunt. They are not calibrated
    against measured fill data; they encode the ordering "fewer listings and
    no sales means you get less", which is the part that is certain. Treat
    a liquidity-adjusted EV as a ranking signal, not a forecast.
    """

    # sell-listing thresholds (supply-based, weaker evidence)
    dead_listings: int = 3
    thin_listings: int = 15
    ok_listings: int = 60

    # 24h sales thresholds (much stronger evidence)
    dead_volume: int = 1
    thin_volume: int = 5
    ok_volume: int = 25

    # what fraction of the listed price you realistically clear
    realization: tuple[float, float, float, float] = (0.55, 0.78, 0.93, 0.98)

    # inputs: you must BUY ten copies, so supply matters directly
    min_listings_to_buy_ten: int = 12

    # concentration guard
    max_ev_share: float = 0.85       # one outcome may not exceed this share of EV
    concentration_exempt_tier: LiquidityTier = LiquidityTier.OK

    def tier(self, listings: Optional[int], volume: Optional[int] = None
             ) -> LiquidityTier:
        """Classify an item. `volume` (24h sales) overrides `listings`."""
        if volume is not None:
            if volume < self.dead_volume:
                return LiquidityTier.DEAD
            if volume < self.thin_volume:
                return LiquidityTier.THIN
            if volume < self.ok_volume:
                return LiquidityTier.OK
            return LiquidityTier.DEEP
        if listings is None:
            # no information at all: assume the worst, do not assume the best
            return LiquidityTier.DEAD
        if listings < self.dead_listings:
            return LiquidityTier.DEAD
        if listings < self.thin_listings:
            return LiquidityTier.THIN
        if listings < self.ok_listings:
            return LiquidityTier.OK
        return LiquidityTier.DEEP

    def realization_factor(self, listings: Optional[int],
                           volume: Optional[int] = None) -> float:
        """Fraction of the listed price you should expect to clear."""
        return self.realization[int(self.tier(listings, volume))]

    def can_buy_ten(self, listings: Optional[int],
                    volume: Optional[int] = None, n: int = 10) -> bool:
        """Is there enough supply to actually assemble the contract?

        With sales volume we can be a bit more generous: an item that sells
        20 a day restocks even if only 6 are listed right now.
        """
        if volume is not None and volume >= n * 2:
            return True
        if listings is None:
            return False
        return listings >= max(self.min_listings_to_buy_ten, n)


@dataclass(slots=True)
class LiquidityReport:
    """Why a recipe was or was not downgraded."""

    ev_raw: float = 0.0              # EV using listed prices
    ev_adjusted: float = 0.0         # EV after realization haircuts
    profit_adjusted: float = 0.0
    roi_adjusted: float = 0.0
    top_ev_share: float = 0.0        # largest single-outcome share of EV
    top_outcome: str = ""
    top_outcome_tier: LiquidityTier = LiquidityTier.DEAD
    worst_tier: LiquidityTier = LiquidityTier.DEEP
    concentrated: bool = False       # EV depends on one illiquid outcome
    unbuyable_inputs: list[str] = None

    def __post_init__(self) -> None:
        if self.unbuyable_inputs is None:
            self.unbuyable_inputs = []

    @property
    def haircut_pct(self) -> float:
        if self.ev_raw <= 0:
            return 0.0
        return (1.0 - self.ev_adjusted / self.ev_raw) * 100.0

    def warnings(self) -> list[str]:
        out: list[str] = []
        if self.concentrated:
            out.append(
                f"{self.top_ev_share*100:.0f}% of EV comes from "
                f"{self.top_outcome} ({self.top_outcome_tier.label}). "
                "The recipe is a bet on that one price being real.")
        if self.worst_tier <= LiquidityTier.DEAD:
            out.append("at least one outcome has effectively no market")
        if self.unbuyable_inputs:
            out.append("not enough listings to buy 10 of: "
                       + ", ".join(self.unbuyable_inputs[:3]))
        return out


def assess(
    outcomes: Sequence,          # tradeup.models.Outcome
    policy: LiquidityPolicy,
    depth: dict[str, tuple[Optional[int], Optional[int]]],
    cost: float,
) -> LiquidityReport:
    """Re-price a contract's outcomes for liquidity.

    `depth` maps market_hash_name -> (sell_listings, 24h_volume). Either may
    be None; volume is preferred when present.
    """
    rep = LiquidityReport()
    if not outcomes:
        return rep

    rep.ev_raw = sum(o.probability * o.net_price for o in outcomes)

    contribs: list[tuple[float, object, LiquidityTier]] = []
    ev_adj = 0.0
    worst = LiquidityTier.DEEP
    for o in outcomes:
        name = o.skin.market_hash_name(o.wear)
        listings, volume = depth.get(name, (None, None))
        tier = policy.tier(listings, volume)
        worst = min(worst, tier)
        adj = o.net_price * policy.realization_factor(listings, volume)
        c = o.probability * adj
        ev_adj += c
        contribs.append((c, o, tier))

    rep.ev_adjusted = ev_adj
    rep.profit_adjusted = ev_adj - cost
    rep.roi_adjusted = (ev_adj / cost) if cost > 0 else 0.0
    rep.worst_tier = worst

    if ev_adj > 0:
        c, o, tier = max(contribs, key=lambda x: x[0])
        rep.top_ev_share = c / ev_adj
        rep.top_outcome = o.skin.market_hash_name(o.wear)
        rep.top_outcome_tier = tier
        rep.concentrated = (
            rep.top_ev_share > policy.max_ev_share
            and tier < policy.concentration_exempt_tier)
    return rep


def breakeven_realization(outcomes: Sequence, cost: float,
                          target: str) -> Optional[float]:
    """What fraction of `target`'s listed price makes the contract break even.

    Answers the only question that matters for a concentrated recipe: how
    wrong can that one price be before this loses money?
    """
    other = sum(o.probability * o.net_price
                for o in outcomes if o.skin.market_hash_name(o.wear) != target)
    hit = [o for o in outcomes if o.skin.market_hash_name(o.wear) == target]
    if not hit:
        return None
    p, net = hit[0].probability, hit[0].net_price
    if p <= 0 or net <= 0:
        return None
    return (cost - other) / (p * net)

"""Core domain models for CS2 trade-up contracts."""
from __future__ import annotations

from dataclasses import dataclass, field
from enum import IntEnum
from typing import Optional

# Global wear brackets. A skin's own Float Cap decides which of these it can
# actually reach -- a 0.10-0.70 skin can never be Factory New.
WEAR_BRACKETS: list[tuple[str, float, float]] = [
    ("Factory New", 0.00, 0.07),
    ("Minimal Wear", 0.07, 0.15),
    ("Field-Tested", 0.15, 0.38),
    ("Well-Worn", 0.38, 0.45),
    ("Battle-Scarred", 0.45, 1.00),
]

WEAR_SHORT = {
    "Factory New": "FN", "Minimal Wear": "MW", "Field-Tested": "FT",
    "Well-Worn": "WW", "Battle-Scarred": "BS",
}


def wear_from_float(f: float) -> str:
    """Map an absolute float value to its wear condition."""
    for name, lo, hi in WEAR_BRACKETS:
        if lo <= f < hi:
            return name
    return "Battle-Scarred"


class Rarity(IntEnum):
    CONSUMER = 0
    INDUSTRIAL = 1
    MILSPEC = 2
    RESTRICTED = 3
    CLASSIFIED = 4
    COVERT = 5
    RARE_SPECIAL = 6  # knives / gloves

    @classmethod
    def parse(cls, raw: str) -> "Rarity":
        key = raw.strip().lower().replace("-", " ").replace("_", " ")
        key = key.replace(" grade", "").replace(" (rare)", "").strip()
        table = {
            "consumer": cls.CONSUMER, "industrial": cls.INDUSTRIAL,
            "mil spec": cls.MILSPEC, "milspec": cls.MILSPEC,
            "restricted": cls.RESTRICTED, "classified": cls.CLASSIFIED,
            "covert": cls.COVERT, "extraordinary": cls.RARE_SPECIAL,
            "contraband": cls.RARE_SPECIAL, "master": cls.RARE_SPECIAL,
            "rare special": cls.RARE_SPECIAL,
        }
        for k, v in table.items():
            if key.startswith(k):
                return v
        raise ValueError(f"Unknown rarity: {raw!r}")

    @property
    def label(self) -> str:
        return {0: "Consumer", 1: "Industrial", 2: "Mil-Spec", 3: "Restricted",
                4: "Classified", 5: "Covert", 6: "Rare Special"}[int(self)]

    def next_tier(self) -> Optional["Rarity"]:
        return Rarity(int(self) + 1) if int(self) < 6 else None


@dataclass(frozen=True, slots=True)
class Skin:
    """A weapon finish (paint kit), independent of wear and quality.

    `min_float`/`max_float` are the skin's Float Cap: the range it can ever
    be generated within. Not every skin spans 0.00-1.00, and the cap decides
    both which wear conditions exist and how the skin behaves in trade-ups.
    """

    id: str
    name: str            # e.g. "AK-47 | Redline"
    rarity: Rarity
    collection: str
    min_float: float     # Best Possible Float
    max_float: float     # Worst Possible Float
    stattrak_capable: bool = True
    souvenir_capable: bool = False

    def market_hash_name(
        self, wear: str, stattrak: bool = False, souvenir: bool = False
    ) -> str:
        """Steam market name for a specific quality + wear.

        StatTrak and Souvenir are mutually exclusive; StatTrak wins if both
        are somehow requested.
        """
        if stattrak:
            prefix = "StatTrak\u2122 "
        elif souvenir:
            prefix = "Souvenir "
        else:
            prefix = ""
        return f"{prefix}{self.name} ({wear})"

    def reachable_wears(self) -> list[str]:
        """Wear conditions this skin can physically exist in, given its cap."""
        return [n for n, lo, hi in WEAR_BRACKETS
                if self.min_float < hi and self.max_float > lo]

    @property
    def float_span(self) -> float:
        return self.max_float - self.min_float

    def clamp(self, f: float) -> float:
        return min(max(f, self.min_float), self.max_float)


@dataclass(slots=True)
class InputItem:
    """One of the n items fed into a contract.

    Since 22 May 2026 a contract may mix Souvenir and normal inputs freely.
    Souvenir attributes are stripped and the OUTPUT is always normal, so
    `souvenir` only affects what this input costs to buy.
    """

    skin: Skin
    float_value: float          # the raw Wear Rating on the item
    price: float = 0.0
    stattrak: bool = False
    souvenir: bool = False

    @property
    def trade_up_float(self) -> float:
        """This item's float rescaled onto the universal 0-1 axis.

        (float - min) / (max - min), using the skin's OWN Float Cap.
        """
        span = self.skin.float_span
        if span <= 0:
            return 0.0
        return (self.float_value - self.skin.min_float) / span

    # legacy alias
    @property
    def normalized(self) -> float:
        return self.trade_up_float

    @property
    def wear(self) -> str:
        return wear_from_float(self.float_value)


@dataclass(slots=True)
class Outcome:
    skin: Skin
    probability: float
    out_float: float            # Outcome Float, in the output skin's own cap
    wear: str
    gross_price: float = 0.0
    net_price: float = 0.0
    priced: bool = True


@dataclass(slots=True)
class ContractResult:
    inputs: list[InputItem] = field(default_factory=list)
    outcomes: list[Outcome] = field(default_factory=list)
    avg_normalized: float = 0.0          # Average Trade-up Float

    cost: float = 0.0
    ev_net: float = 0.0
    profit: float = 0.0
    roi: float = 0.0
    profit_chance: float = 0.0
    stddev: float = 0.0
    unpriced_mass: float = 0.0
    stattrak: bool = False

    @property
    def avg_trade_up_float(self) -> float:
        return self.avg_normalized

    @property
    def n_inputs(self) -> int:
        return len(self.inputs)

    @property
    def n_souvenir_inputs(self) -> int:
        return sum(1 for i in self.inputs if i.souvenir)

"""Budget-constrained search for profitable trade-up recipes.

The search space is huge (every combination of 10 inputs across every
collection and wear), so this uses a structured heuristic instead of brute
force -- the same structure real profitable contracts have:

    k inputs from a TARGET collection (the one with valuable outputs)
  + (N-k) FILLER inputs, chosen to be as cheap as possible and, ideally,
    to drag the normalized float average DOWN.

Since 22 May 2026 a purchasable "unit" is a (skin, wear, quality) triple,
where quality is normal or Souvenir. Souvenir copies are frequently the
cheapest way to fill a slot -- especially at Consumer and Industrial grade --
and the souvenir attributes are stripped by the contract anyway. The OUTPUT
is always normal quality.

Every candidate returned by `search` has been fully priced and scored by the
real engine; nothing is estimated at the end.
"""
from __future__ import annotations

from dataclasses import dataclass
from typing import Callable, Iterable, Optional

from .data.loader import SkinDB
from .engine import OddsModel, evaluate_contract, n_contracts_for_confidence
from .models import ContractResult, InputItem, Rarity, Skin
from .pricing import PriceService, sourceable_float


def contract_size(rarity: Rarity) -> int:
    """10 for normal contracts; Covert -> gold pool contracts take 5."""
    return 5 if rarity == Rarity.COVERT else 10


@dataclass(slots=True)
class Unit:
    """A purchasable input: a specific skin, wear and quality."""

    skin: Skin
    wear: str
    price: float
    float_value: float
    souvenir: bool = False

    @property
    def normalized(self) -> float:
        span = self.skin.max_float - self.skin.min_float
        return 0.0 if span <= 0 else (self.float_value - self.skin.min_float) / span

    @property
    def label(self) -> str:
        return f"{'Souvenir ' if self.souvenir else ''}{self.skin.name} ({self.wear})"


@dataclass(slots=True)
class Candidate:
    result: ContractResult
    target_collection: str
    filler_collection: Optional[str]
    k_target: int

    @property
    def score(self) -> float:
        return self.result.profit


def build_units(
    db: SkinDB,
    prices: PriceService,
    rarity: Rarity,
    *,
    float_percentile: float = 0.15,
    stattrak: bool = False,
    souvenir: bool = True,
    max_unit_price: Optional[float] = None,
) -> dict[str, list[Unit]]:
    """All purchasable inputs at `rarity`, grouped by collection.

    Only collections that actually have a higher tier are kept -- a skin from
    a dead-end collection is pure filler and contributes no outcome mass.

    `souvenir=True` additionally prices the Souvenir variant of every
    souvenir-capable skin and keeps it as a separate unit. StatTrak
    contracts cannot include Souvenir items, so the flag is forced off.
    """
    if stattrak:
        souvenir = False

    units: dict[str, list[Unit]] = {}
    for coll in db.tradeable_collections(rarity):
        for skin in db.pool(coll, rarity):
            if stattrak and not skin.stattrak_capable:
                continue
            for wear in skin.reachable_wears():
                f = sourceable_float(skin, wear, float_percentile)
                if f is None:
                    continue
                qualities = [False]
                if souvenir and skin.souvenir_capable:
                    qualities.append(True)
                for sv in qualities:
                    p = prices.buy_price(skin, wear, stattrak, sv)
                    if p is None or p <= 0:
                        continue
                    if max_unit_price is not None and p > max_unit_price:
                        continue
                    units.setdefault(coll, []).append(Unit(skin, wear, p, f, sv))
    for coll in units:
        units[coll].sort(key=lambda u: (u.price, u.normalized))
    return units


def _cheapest_low_float(units: list[Unit], n: int) -> list[Unit]:
    """Cheapest n units, tie-broken toward the lowest normalized float.

    Repeats are allowed: you can buy ten copies of the same cheap listing.
    """
    if not units:
        return []
    best = min(units, key=lambda u: (u.price, u.normalized))
    return [best] * n


def _best_float_efficient(units: list[Unit], n: int, budget: float) -> list[Unit]:
    """n copies of the unit with the lowest normalized float we can afford."""
    affordable = [u for u in units if u.price * n <= budget]
    if not affordable:
        return []
    best = min(affordable, key=lambda u: (u.normalized, u.price))
    return [best] * n


def search(
    db: SkinDB,
    prices: PriceService,
    rarity: Rarity,
    *,
    budget: float,
    odds_model: OddsModel = "outcome_weighted",
    fee_rate: Optional[float] = None,
    stattrak: bool = False,
    souvenir: bool = True,
    float_percentile: float = 0.15,
    min_roi: float = 1.0,
    max_unpriced_mass: float = 0.10,
    top_n: int = 25,
    max_filler_collections: int = 8,
    progress: Optional[Callable[[str], None]] = None,
) -> list[Candidate]:
    """Enumerate promising recipes at `rarity` under `budget` (total input cost)."""
    n_inputs = contract_size(rarity)
    if rarity.next_tier() is None:
        return []

    units = build_units(
        db, prices, rarity, float_percentile=float_percentile,
        stattrak=stattrak, souvenir=souvenir, max_unit_price=budget)
    if not units:
        return []

    # Good filler candidates are collections whose cheapest input is nearly free
    filler_ranked = sorted(units, key=lambda c: min(u.price for u in units[c]))
    fillers = filler_ranked[:max_filler_collections]

    pool, pfn = db.pool, prices.price_fn()
    out: list[Candidate] = []
    targets = list(units)

    for ti, target in enumerate(targets):
        if progress and ti % 10 == 0:
            progress(f"{ti}/{len(targets)} collections")
        t_units = units[target]

        # 1) pure single-collection contract
        combos: list[tuple[list[Unit], Optional[str], int]] = []
        pure = _cheapest_low_float(t_units, n_inputs)
        if pure:
            combos.append((pure, None, n_inputs))
        lowf = _best_float_efficient(t_units, n_inputs, budget)
        if lowf and lowf != pure:
            combos.append((lowf, None, n_inputs))

        # 2) target + filler mixes
        for filler in fillers:
            if filler == target:
                continue
            f_units = units[filler]
            for k in range(1, n_inputs):
                t_part = _cheapest_low_float(t_units, k)
                f_part = _cheapest_low_float(f_units, n_inputs - k)
                if not t_part or not f_part:
                    continue
                if sum(u.price for u in t_part + f_part) > budget:
                    continue
                combos.append((t_part + f_part, filler, k))

                # float-optimized variant: spend the remaining budget on
                # lower-float fillers, pulling the output wear down
                rest = budget - sum(u.price for u in t_part)
                f_low = _best_float_efficient(f_units, n_inputs - k, rest)
                if f_low and f_low != f_part:
                    combos.append((t_part + f_low, filler, k))

        for units_combo, filler, k in combos:
            cost = sum(u.price for u in units_combo)
            if cost > budget or cost <= 0:
                continue
            inputs = [InputItem(u.skin, u.float_value, u.price, stattrak, u.souvenir)
                      for u in units_combo]
            res = evaluate_contract(
                inputs, pool, pfn, odds_model=odds_model, fee_rate=fee_rate,
                stattrak=stattrak, skin_by_id=db.by_id)
            if res.unpriced_mass > max_unpriced_mass or res.roi < min_roi:
                continue
            out.append(Candidate(res, target, filler, k))

    # de-duplicate identical input baskets, keep the best
    seen: dict[tuple, Candidate] = {}
    for c in out:
        key = tuple(sorted((i.skin.id, round(i.float_value, 5), i.souvenir)
                           for i in c.result.inputs))
        if key not in seen or c.score > seen[key].score:
            seen[key] = c
    return sorted(seen.values(), key=lambda c: c.score, reverse=True)[:top_n]


def robust_filter(
    candidates: Iterable[Candidate],
    db: SkinDB,
    prices: PriceService,
    *,
    fee_rate: Optional[float] = None,
) -> list[Candidate]:
    """Keep only candidates that stay profitable under BOTH odds models.

    Valve does not publish the outcome-selection rule and community tools
    disagree. A recipe profitable under only one interpretation is a
    modelling artifact, not an edge.
    """
    kept, pfn = [], prices.price_fn()
    for c in candidates:
        alt = evaluate_contract(
            c.result.inputs, db.pool, pfn, odds_model="per_input",
            fee_rate=fee_rate, stattrak=c.result.stattrak, skin_by_id=db.by_id)
        if alt.profit > 0 and c.result.profit > 0:
            kept.append(c)
    return kept

"""Glue between a price source and the trade-up engine."""
from __future__ import annotations

from dataclasses import dataclass
from typing import Optional, Protocol

from .models import WEAR_BRACKETS, Skin
from .prices.steam import Quote


class PriceSource(Protocol):
    """Anything that can answer 'what is this market_hash_name worth?'.

    The engine never talks to Steam directly, so swapping in another
    marketplace is a matter of implementing this one method.
    """

    def get(self, market_hash_name: str) -> Optional[Quote]: ...


@dataclass(slots=True)
class PricingPolicy:
    """How raw market quotes become the numbers used in the EV model.

    min_volume:
        Ignore OUTPUT quotes thinner than this. An "EV-positive" trade-up
        whose jackpot skin sells twice a week is not a real edge -- you will
        sit on inventory or dump below median.
    min_input_volume:
        Same idea for INPUTS, but stricter: you need to buy ten copies.
        Souvenir listings in particular are often near-unique, so a R$0.40
        quote backed by two sales is not a price you can act on.
    buy_basis / sell_basis:
        'lowest' is what you pay to acquire an input right now.
        'median' is what you can realistically get out for. Using `lowest`
        on both sides is the most common way these tools lie to you.
    haircut:
        Extra safety margin applied to output prices (0.05 = assume you sell
        5% under the reference price).

    NOTE: after `bulk-warm`, quotes carry `lowest` + a LISTING COUNT in the
    volume column and no median. `sell_price` then falls back to `best`.
    Confirm a shortlist with `warm-prices` before committing money.
    """

    min_volume: int = 3
    min_input_volume: int = 10
    buy_basis: str = "lowest"
    sell_basis: str = "median"
    haircut: float = 0.03


class PriceService:
    def __init__(self, source: PriceSource, policy: PricingPolicy | None = None):
        self.source = source
        self.policy = policy or PricingPolicy()

    def _quote(self, name: str) -> Optional[Quote]:
        return self.source.get(name)

    def buy_price(
        self, skin: Skin, wear: str, stattrak: bool = False, souvenir: bool = False
    ) -> Optional[float]:
        """Cost to acquire one copy. `souvenir=True` prices the Souvenir variant."""
        q = self._quote(skin.market_hash_name(wear, stattrak, souvenir))
        if q is None:
            return None
        if q.volume is not None and q.volume < self.policy.min_input_volume:
            return None  # cannot reliably buy ten of these
        return getattr(q, self.policy.buy_basis) or q.best

    def sell_price(
        self, skin: Skin, wear: str, stattrak: bool = False
    ) -> Optional[float]:
        """Proceeds for the OUTPUT, which is always normal quality."""
        q = self._quote(skin.market_hash_name(wear, stattrak, souvenir=False))
        if q is None:
            return None
        if q.volume is not None and q.volume < self.policy.min_volume:
            return None
        v = getattr(q, self.policy.sell_basis) or q.best
        return None if v is None else v * (1.0 - self.policy.haircut)

    # engine-compatible callback
    def price_fn(self):
        return lambda skin, wear, st: self.sell_price(skin, wear, st)


def wear_window(skin: Skin, wear: str) -> Optional[tuple[float, float]]:
    """Intersection of a wear bracket with the skin's own float range."""
    b = next((x for x in WEAR_BRACKETS if x[0] == wear), None)
    if b is None:
        return None
    lo, hi = max(b[1], skin.min_float), min(b[2], skin.max_float)
    return (lo, hi) if hi > lo else None


def sourceable_float(skin: Skin, wear: str, percentile: float = 0.15) -> Optional[float]:
    """The float you can realistically buy inside a wear tier.

    `percentile=0.15` means "with some listing hunting I can find an item in
    the bottom 15% of this wear bracket". Set it to 0.0 only if you are
    actually willing to pay float-sniper premiums, and 0.5 for a lazy
    market-order assumption.
    """
    w = wear_window(skin, wear)
    if w is None:
        return None
    lo, hi = w
    return lo + percentile * (hi - lo)

def depth_map(names: set[str], currency: int) -> dict[str, tuple]:
    """market_hash_name -> (sell_listings, 24h_volume) straight from cache.

    With bulk data the `volume` column holds SELL LISTINGS (supply). Once
    warm-prices has run, `median` is set and the same column is real 24h
    sales. The two are distinguished here so the liquidity model can weight
    them correctly.
    """
    import sqlite3

    out: dict[str, tuple] = {}
    if not CACHE_PATH.exists() or not names:
        return out
    conn = sqlite3.connect(str(CACHE_PATH))
    try:
        qs = list(names)
        for i in range(0, len(qs), 400):
            chunk = qs[i:i + 400]
            marks = ",".join("?" * len(chunk))
            for nm, vol, med in conn.execute(
                    f"SELECT name, volume, median FROM prices "
                    f"WHERE currency=? AND name IN ({marks})",
                    [currency, *chunk]):
                out[nm] = (None, vol) if med is not None else (vol, None)
    except Exception:
        pass
    finally:
        conn.close()
    return out


CUR_ID = {"USD": 1, "GBP": 2, "EUR": 3, "BRL": 7, "RUB": 5, "CNY": 23}


def run_scan(p: dict[str, Any]) -> list[dict]:
    key = hashlib.md5(json.dumps(p, sort_keys=True).encode()).hexdigest()
    hit = _scan_cache.get(key)
    if hit and time.time() - hit[0] < SCAN_TTL:
        return hit[1]

    d = db()
    r = RARITIES[p["rarity"]]
    policy = PricingPolicy(min_volume=p["min_volume"],
                           min_input_volume=p["min_input_volume"],
                           haircut=p["haircut"])
    prices = price_service(policy, p["currency"])

    cands = search(d, prices, r,
                   budget=p["budget"],
                   odds_model=p["odds"],
                   fee_rate=p["fee_rate"],
                   souvenir=p["souvenir"],
                   float_percentile=p["float_percentile"],
                   min_roi=0.0,               # filter later, in the UI
                   top_n=400)

    rows = [explain(c) for c in cands]
    img = images()
    cur = CUR_ID.get(p["currency"].upper(), 1)

    # attach float caps, images and liquidity
    want: set[str] = set()
    for row in rows:
        for o in row["outcomes"]:
            want.add(f"{o['skin']} ({o['wear']})")
    depth = depth_map(want, cur)
    pol = LiquidityPolicy() if HAVE_LIQUIDITY else None

    for row in rows:
        for o in row["outcomes"]:
            s = d.by_name(o["skin"])
            o["min_float"] = round(s.min_float, 4) if s else None
            o["max_float"] = round(s.max_float, 4) if s else None
            o["image"] = img.get(o["skin"])
            nm = f"{o['skin']} ({o['wear']})"
            listings, vol = depth.get(nm, (None, None))
            o["listings"] = listings
            o["volume24"] = vol
            if pol:
                o["liquidity"] = pol.tier(listings, vol).label
        for i in row["inputs"]:
            s = d.by_name(i["skin"])
            i["min_float"] = round(s.min_float, 4) if s else None
            i["max_float"] = round(s.max_float, 4) if s else None
            i["image"] = img.get(i["skin"])

        if HAVE_LIQUIDITY:
            class _O:  # tiny shim so assess() can read the dicts
                __slots__ = ("skin", "probability", "wear", "net_price")

                def __init__(self, o):
                    self.skin = type("S", (), {
                        "market_hash_name": staticmethod(
                            lambda w, st=False, sv=False, n=o["skin"]: f"{n} ({w})")})
                    self.probability = o["p_pct"] / 100.0
                    self.wear = o["wear"]
                    self.net_price = o["net"]

            shim = [_O(o) for o in row["outcomes"]]
            rep = assess(shim, pol, depth, row["cost"])
            row["ev_adjusted"] = round(rep.ev_adjusted, 2)
            row["profit_adjusted"] = round(rep.profit_adjusted, 2)
            row["roi_adjusted_pct"] = round(rep.roi_adjusted * 100, 2)
            row["top_ev_share_pct"] = round(rep.top_ev_share * 100, 1)
            row["top_outcome"] = rep.top_outcome
            row["concentrated"] = rep.concentrated
            row["worst_liquidity"] = rep.worst_tier.label
            row["warnings"] = rep.warnings()
            be = breakeven_realization(shim, row["cost"], rep.top_outcome)
            row["breakeven_pct"] = round(be * 100, 1) if be else None
        else:
            row["ev_adjusted"] = row["ev_net"]
            row["profit_adjusted"] = row["profit"]
            row["roi_adjusted_pct"] = row["roi_pct"]
            row["warnings"] = []
            row["concentrated"] = False

    _scan_cache[key] = (time.time(), rows)
    return rows

            {"key": k, "label": v.label,
             "collections": len(d.tradeable_collections(v)),
             "outputs": len(d.tradeable_collections(v.next_tier()))
                        if v.next_tier() else 0}
            for k, v in RARITIES.items() if v.next_tier()
        ],
    }


@app.get("/api/scan")
def api_scan(
    budget: float = Query(200.0, gt=0),
    rarity: str = "milspec",
    currency: str = "BRL",
    fee_rate: Optional[float] = Query(None, description="blank = exact Steam model"),
    souvenir: bool = True,
    float_percentile: float = 0.15,
    min_volume: int = 3,
    min_input_volume: int = 10,
    haircut: float = 0.03,
    odds: Literal["per_input", "outcome_weighted"] = "per_input",
    # UI-side filters
    min_roi: float = 100.0,
    min_profit: float = 0.0,
    max_cost: Optional[float] = None,
    liquidity: Literal["any", "ok", "deep"] = "any",
    hide_concentrated: bool = False,
    sort: Literal["profit", "roi", "ev", "cost", "safety"] = "profit",
    adjusted: bool = True,
    top: int = 20,
):
    if rarity not in RARITIES:
        raise HTTPException(400, f"bad rarity: {rarity}")

    rows = run_scan({
        "budget": budget, "rarity": rarity, "currency": currency,
        "fee_rate": fee_rate, "souvenir": souvenir,
        "float_percentile": float_percentile, "min_volume": min_volume,
        "min_input_volume": min_input_volume, "haircut": haircut, "odds": odds,
    })

    pk = "profit_adjusted" if adjusted else "profit"
    rk = "roi_adjusted_pct" if adjusted else "roi_pct"
    ek = "ev_adjusted" if adjusted else "ev_net"

    out = []
    rank = {"dead": 0, "thin": 1, "ok": 2, "deep": 3, "unknown": 0}
    for r in rows:
        if r[rk] < min_roi or r[pk] < min_profit:
            continue
        if max_cost is not None and r["cost"] > max_cost:
            continue
        if hide_concentrated and r.get("concentrated"):
            continue
        if liquidity != "any":
            need = rank["ok"] if liquidity == "ok" else rank["deep"]
            if rank.get(r.get("worst_liquidity", "unknown"), 0) < need:
                continue
        out.append(r)

    keys = {"profit": pk, "roi": rk, "ev": ek, "cost": "cost",
            "safety": "profit_chance_pct"}
    rev = sort != "cost"
    out.sort(key=lambda r: r.get(keys[sort], 0), reverse=rev)

    return {
        "count": len(out),
        "scanned": len(rows),
        "currency": currency,
        "symbol": CURRENCY_SYMBOL.get(currency.upper(), ""),
        "adjusted": adjusted and HAVE_LIQUIDITY,
        "liquidity_available": HAVE_LIQUIDITY,
        "results": out[:top],
    }

"""Re-score scan results for liquidity, using your existing price cache.

    python -m tradeup.cli scan ... --json-out results.json
    python check_liquidity.py results.json --currency BRL

Standalone and read-only: it reads data/prices.sqlite directly, so it works
before you patch anything.

A trade-up EV built on the lowest listing assumes you can sell instantly at
that price. For a popular Field-Tested rifle that is roughly true. For an
unpopular skin in Well-Worn it is fiction -- three listings, no recent
sales, and a price one optimistic seller typed in.

This reports two things the EV number hides:

  REALIZATION   what you should expect to clear, given how thin the book is
  CONCENTRATION how much of the EV rides on a single illiquid outcome

Plus the only question that matters for a concentrated recipe: how wrong
can that one price be before the contract loses money?
"""
from __future__ import annotations

import argparse
import json
import sqlite3
import sys
from pathlib import Path

CURRENCY = {"USD": 1, "GBP": 2, "EUR": 3, "BRL": 7, "RUB": 5, "CNY": 23}
CUR_SYM = {1: "$", 2: "£", 3: "€", 7: "R$", 5: "₽", 23: "¥"}

# (max_listings, label, realization factor)
TIERS = [(3, "dead", 0.55), (15, "thin", 0.78), (60, "ok", 0.93),
         (10 ** 9, "deep", 0.98)]
VOL_TIERS = [(1, "dead", 0.55), (5, "thin", 0.78), (25, "ok", 0.93),
             (10 ** 9, "deep", 0.98)]

MAX_EV_SHARE = 0.85       # above this, one outcome dominates the recipe
MIN_LISTINGS_BUY = 12     # need to buy ten copies


def classify(listings, volume):
    """-> (label, realization). 24h sales beat supply when available."""
    if volume is not None:
        for cap, label, f in VOL_TIERS:
            if volume < cap:
                return label, f
    if listings is None:
        return "unknown", 0.55
    for cap, label, f in TIERS:
        if listings < cap:
            return label, f
    return "deep", 0.98


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("results", help="JSON from scan --json-out")
    ap.add_argument("--currency", default="USD")
    ap.add_argument("--cache", default="data/prices.sqlite")
    ap.add_argument("--top", type=int, default=10)
    ap.add_argument("--max-ev-share", type=float, default=MAX_EV_SHARE)
    a = ap.parse_args()

    cur = CURRENCY.get(a.currency.upper(), 1)
    sym = CUR_SYM.get(cur, "")

    rp, cp = Path(a.results), Path(a.cache)
    if not rp.exists():
        print(f"! {rp} not found. Run scan with --json-out {rp} first.")
        sys.exit(1)
    if not cp.exists():
        print(f"! {cp} not found.")
        sys.exit(1)

    rows = json.loads(rp.read_text(encoding="utf-8"))
    db = sqlite3.connect(str(cp))

    def depth(name: str):
        r = db.execute(
            "SELECT volume, median FROM prices WHERE name=? AND currency=?",
            (name, cur)).fetchone()
        if not r:
            return None, None
        vol, med = r
        # with no median the cache came from bulk-warm: `volume` holds
        # SELL LISTINGS (supply), not 24h sales
        return (None, vol) if med is not None else (vol, None)

    print("=" * 100)
    print(f"LIQUIDITY RE-SCORE   {len(rows)} recipes   currency {a.currency.upper()}")
    print("=" * 100)
    print("\nEV assumes you sell at the listed price. These numbers assume you")
    print("have to find a real buyer.\n")

    hdr = (f"{'#':>3} {'target collection':<28}{'cost':>9}{'EV raw':>9}"
           f"{'EV adj':>9}{'profit':>9}{'ROI':>8}  {'top outcome share':<20}{'verdict'}")
    print(hdr)
    print("-" * 100)

    keep = []
    for i, r in enumerate(rows[:a.top], 1):
        cost = r["cost"]
        outs = r["outcomes"]
        ev_raw = sum(o["p_pct"] / 100 * o["net"] for o in outs)

        ev_adj, contribs, worst = 0.0, [], "deep"
        order = {"dead": 0, "unknown": 0, "thin": 1, "ok": 2, "deep": 3}
        for o in outs:
            name = f"{o['skin']} ({o['wear']})"
            listings, vol = depth(name)
            label, f = classify(listings, vol)
            if order[label] < order[worst]:
                worst = label
            c = o["p_pct"] / 100 * o["net"] * f
            ev_adj += c
            contribs.append((c, name, label, listings, vol, o))

        # can you even buy the inputs? need 10 copies of each distinct skin
        short = []
        for key in {x["skin"] for x in r["inputs"]}:
            need = sum(1 for x in r["inputs"] if x["skin"] == key)
            best = None
            for nm, vol, med in db.execute(
                    "SELECT name, volume, median FROM prices WHERE name LIKE ? "
                    "AND currency=?", (f"{key} (%", cur)):
                if best is None or (vol or 0) > (best[0] or 0):
                    best = (vol, med)
            if best is None:
                continue
            listings = None if best[1] is not None else best[0]
            vol24 = best[0] if best[1] is not None else None
            if vol24 is not None and vol24 >= need * 2:
                continue
            if listings is not None and listings < max(MIN_LISTINGS_BUY, need):
                short.append((key, listings, need))

        contribs.sort(reverse=True, key=lambda x: x[0])
        share = contribs[0][0] / ev_adj if ev_adj > 0 else 0.0
        top_label = contribs[0][2]
        profit = ev_adj - cost
        roi = ev_adj / cost if cost else 0
        concentrated = share > a.max_ev_share and top_label in ("dead", "thin", "unknown")

        verdict = ("REJECT" if profit <= 0 else
                   "NO SUPPLY" if short else
                   "RISKY" if concentrated else "ok")
        if verdict not in ("REJECT", "NO SUPPLY"):
            keep.append((i, r, contribs, ev_adj, profit, roi, concentrated))

        print(f"{i:>3} {r['target_collection'][:27]:<28}{cost:>9.2f}{ev_raw:>9.2f}"
              f"{ev_adj:>9.2f}{profit:>+9.2f}{roi*100:>7.1f}%  "
              f"{share*100:>5.1f}% {top_label:<13} {verdict}")
        for key, listed, need in short[:2]:
            print(f"      ! only {listed} listed of {key[:40]} -- need {need}")

    # ---- detail on the best surviving recipe --------------------------
    print("\n" + "=" * 100)
    if not keep:
        print("Nothing survives a liquidity haircut. Every recipe above depends")
        print("on selling into a market that may not absorb it.")
        db.close()
        return

    i, r, contribs, ev_adj, profit, roi, conc = keep[0]
    print(f"DETAIL  recipe #{i}  {r['target_collection']}")
    print("=" * 100)
    print(f"\n{'outcome':<40}{'P':>6}{'net':>10}{'listed':>8}{'24h':>6}"
          f"{'tier':>7}{'EV share':>10}")
    for c, name, label, listings, vol, o in contribs:
        sh = c / ev_adj * 100 if ev_adj else 0
        print(f"{name[:39]:<40}{o['p_pct']:>5.1f}%{o['net']:>10.2f}"
              f"{(listings if listings is not None else '-'):>8}"
              f"{(vol if vol is not None else '-'):>6}{label:>7}{sh:>9.1f}%")

    # break-even on the dominant outcome
    top_c, top_name, top_label, _, _, top_o = contribs[0]
    other = sum(o["p_pct"] / 100 * o["net"] for o in r["outcomes"]
                if f"{o['skin']} ({o['wear']})" != top_name)
    p, net = top_o["p_pct"] / 100, top_o["net"]
    if p > 0 and net > 0:
        be = (r["cost"] - other) / (p * net)
        print(f"\nBreak-even: you need {be*100:.1f}% of the listed price on")
        print(f"  {top_name}")
        if be > 0.9:
            print("  -> almost no margin for error. One stale listing and this loses.")
        elif be > 0.7:
            print("  -> some cushion, but verify that price manually.")
        else:
            print("  -> comfortable cushion.")

    print("\nInputs to buy (10 copies each -- check supply):")
    seen = set()
    for inp in r["inputs"]:
        wear = None
        for o in r["outcomes"]:
            pass
        key = inp["skin"]
        if key in seen:
            continue
        seen.add(key)
        n = sum(1 for x in r["inputs"] if x["skin"] == key)
        # find the listing count for any wear of this input
        best = None
        for row in db.execute(
                "SELECT name, volume, median FROM prices WHERE name LIKE ? "
                "AND currency=?", (f"{key} (%", cur)):
            nm, vol, med = row
            if best is None or (vol or 0) > (best[1] or 0):
                best = (nm, vol, med)
        if best:
            listings, vol = (None, best[1]) if best[2] is not None else (best[1], None)
            label, _ = classify(listings, vol)
            flag = ("  <- too few listed to buy 10"
                    if listings is not None and listings < MIN_LISTINGS_BUY else "")
            print(f"  {n:>2}x {key[:44]:<46}{sym}{inp['price']:>8.2f}  "
                  f"{label:<6}{flag}")
        else:
            print(f"  {n:>2}x {key[:44]:<46}{sym}{inp['price']:>8.2f}  no depth data")

    print("\n" + "=" * 100)
    print("Realization factors are blunt (dead 55%, thin 78%, ok 93%, deep 98%).")
    print("They encode 'fewer listings and no sales means you get less', which is")
    print("certain -- not a calibrated forecast. Use this to RANK, then confirm")
    print("the survivors with warm-prices for real 24h sales volume.")
    print("=" * 100)
    db.close()


if __name__ == "__main__":

"""Standalone Steam Market diagnostic. Run from the project root:

    python diagnose_steam.py --currency BRL

Answers three questions a progress bar cannot:
  1. Is Steam responding to *this* IP at all, and how fast?
  2. Is it rate-limiting (429) or soft-banning (persistent 429 / empty 200)?
  3. How far along is the cache, and is it still growing?

Safe to run while a warm-prices/bulk-warm job is going: it makes at most a
handful of requests and only reads the cache.
"""
from __future__ import annotations

import argparse
import sqlite3
import time
from pathlib import Path

import httpx

CURRENCY = {"USD": 1, "GBP": 2, "EUR": 3, "BRL": 7, "RUB": 5, "CNY": 23}
URL = "https://steamcommunity.com/market/priceoverview/"

# Cheap, extremely liquid items that definitely exist on the market
PROBES = [
    "AK-47 | Elite Build (Field-Tested)",
    "MP9 | Storm (Field-Tested)",
    "P250 | Sand Dune (Field-Tested)",
]


def probe(currency: int, timeout: float, n: int) -> None:
    client = httpx.Client(
        timeout=httpx.Timeout(timeout, connect=10.0),
        headers={"User-Agent": "cs2-tradeup-lab/1.0 (diagnostic)"},
    )
    print(f"\n--- Live requests (timeout={timeout}s) ---")
    for name in PROBES[:n]:
        t0 = time.monotonic()
        try:
            r = client.get(URL, params={
                "appid": 730, "currency": currency, "market_hash_name": name})
            dt = time.monotonic() - t0
            ra = r.headers.get("Retry-After")
            body = (r.text or "")[:90].replace("\n", " ")
            print(f"  {dt:6.2f}s  HTTP {r.status_code}"
                  f"{f'  Retry-After={ra}' if ra else ''}  {name}")
            print(f"          {body}")
            if r.status_code == 429:
                print("          -> RATE LIMITED. Stop all requests and wait;")
                print("             retrying while banned can extend the ban.")
            elif r.status_code == 200 and '"success":true' not in r.text.lower():
                print("          -> 200 but not a successful quote (soft-ban shape).")
        except httpx.TimeoutException:
            print(f"  {timeout:6.2f}s  TIMEOUT   {name}")
            print("          -> Steam accepted the connection but never replied.")
        except httpx.HTTPError as e:
            print(f"  {time.monotonic()-t0:6.2f}s  {type(e).__name__}: {e}")
        time.sleep(4)
    client.close()


def cache_report(path: Path, currency: int) -> None:
    print(f"\n--- Cache ({path}) ---")
    if not path.exists():
        print("  no cache file yet")
        return
    db = sqlite3.connect(str(path))
    total = db.execute("SELECT count(*) FROM prices WHERE currency=?",
                       (currency,)).fetchone()[0]
    priced = db.execute(
        "SELECT count(*) FROM prices WHERE currency=? AND median IS NOT NULL",
        (currency,)).fetchone()[0]
    print(f"  rows: {total}   with median: {priced}")
    if total and not priced:
        print("  (no medians -> this cache came from bulk-warm, which is expected)")
    row = db.execute(
        "SELECT name, ts FROM prices WHERE currency=? ORDER BY ts DESC LIMIT 1",
        (currency,)).fetchone()
    if row:
        age = time.time() - row[1]
        print(f"  newest: {row[0]}")
        print(f"  written {age:.0f}s ago"
              + ("   <-- STALLED: nothing new is being cached"
                 if age > 120 else "   (actively writing)"))
    recent = db.execute("SELECT count(*) FROM prices WHERE currency=? AND ts > ?",
                        (currency, time.time() - 300)).fetchone()[0]
    print(f"  written in last 5 min: {recent}  (~{recent/5:.1f}/min)")
    try:
        misses = db.execute("SELECT count(*) FROM misses WHERE currency=?",
                            (currency,)).fetchone()[0]
        print(f"  known-empty names cached: {misses}")
    except sqlite3.OperationalError:
        pass


def main() -> None:
    ap = argparse.ArgumentParser()
    ap.add_argument("--currency", default="USD")
    ap.add_argument("--cache", default="data/prices.sqlite")
    ap.add_argument("--timeout", type=float, default=20.0)
    ap.add_argument("--requests", type=int, default=2)
    a = ap.parse_args()
    cur = CURRENCY.get(a.currency.upper(), 1)
    print(f"Currency {a.currency.upper()} (id {cur})")
    cache_report(Path(a.cache), cur)
    probe(cur, a.timeout, a.requests)
    print("\nIf live requests succeed but the cache is stalled, the running job "
          "is stuck in a backoff sleep -- restart it, preferably with bulk-warm.")


if __name__ == "__main__":
    main()

    print(f"\n3. COVERAGE  (share of market names your scans can price)")
    print(f"   {'tier':<12}{'have':>7}{'need':>7}{'':>3}{'':<30}")
    tier_missing: dict[int, list[str]] = {}
    for t in sorted(want_tiers):
        names: set[str] = set()
        for s in skins:
            if s["rarity"] != t:
                continue
            for w in reachable_wears(s["min_float"], s["max_float"]):
                names.add(hash_name(s["name"], w))
                if a.souvenir and s.get("souvenir_capable"):
                    names.add(hash_name(s["name"], w, souvenir=True))
        if not names:
            continue
        have = names & cached
        tier_missing[t] = sorted(names - have)
        frac = len(have) / len(names)
        print(f"   {TIERS[t]:<12}{len(have):>7}{len(names):>7}  "
              f"{bar(frac)} {frac*100:5.1f}%")

    # a tier is only scannable if its OUTPUT tier is priced too
    def coverage(tier: int) -> tuple[int, int]:
        names = set()
        for s in skins:
            if s["rarity"] != tier:
                continue
            for w in reachable_wears(s["min_float"], s["max_float"]):
                names.add(hash_name(s["name"], w))
        return len(names & cached), len(names)

    print(f"\n   scan readiness (a tier needs its OUTPUT tier priced too):")
    for t in sorted(want_tiers):
        nxt = t + 1
        if nxt not in TIERS or t not in tier_missing:
            continue
        have_o, need_o = coverage(nxt)
        if not need_o:
            continue
        ofrac = have_o / need_o
        verdict = ("READY" if ofrac > 0.8 else
                   "partial" if ofrac > 0.3 else "not yet")
        print(f"     {TIERS[t]:<12} -> {TIERS[nxt]:<12} "
              f"outputs {ofrac*100:5.1f}%   {verdict}")

    # -- 4. what is missing ----------------------------------------------
    if a.missing and a.rarity:
        t = want_tiers[0]
        miss = tier_missing.get(t, [])
        print(f"\n4. MISSING from {TIERS[t]} ({len(miss)} names, showing "
              f"{min(a.missing, len(miss))})")
        for n in miss[:a.missing]:
            print(f"     {n}")
        if miss:
            print("\n   Missing names usually have no live listing at all.")
            print("   Confirm one by searching it on the Steam market.")

    print("\n" + "=" * 68)
    if med:
        print("Median data present -- volume filters mean 24h SALES, as intended.")
    else:
        print("No medians: this is bulk data, so the volume column holds")
        print("SELL LISTINGS (supply), not 24h sales.")
        print("  --min-input-volume 10  still reads well: 'at least 10 listed',")
        print("                         which is exactly what you need to buy 10.")
        print("  --min-volume 3         is weaker than intended for outputs; it")
        print("                         means 3 listed, not 3 sold per day.")
        print("Confirm a shortlist with warm-prices before committing money.")
    print("=" * 68)
    db.close()


if __name__ == "__main__":
    main()

# Liquidity: the gap between a listed price and a sale

## Your #1 result, decomposed

```
M4A4 | Modern Hunter (WW)      50%   net 236.19   ->  99.6% of the EV
M4A4 | Sheet Lightning (FT)    25%   net   1.00   ->   0.2%
Glock-18 | Glockingbird (FT)   25%   net   0.81   ->   0.2%
```

Strip the jackpot and EV collapses from 118.55 to **0.45** against a cost of
101.80. That is not a trade-up with variance -- it is a single bet that one
Well-Worn price is real.

Break-even needs **85.8%** of that listed price. A Well-Worn skin nobody
wants does not clear 85.8% of an optimistic listing.

Sensitivity:

| you realize | EV | profit | ROI |
|---|---|---|---|
| 100% of listed | 118.55 | +16.75 | 116.5% |
| 80% | 94.93 | −6.87 | 93.2% |
| 60% | 71.31 | −30.49 | 70.0% |

The reference tool in your screenshot shows the same thing from the other
side: at its API price the contract is −47.65, and only becomes +37.45 when
the output price is overridden upward. Same contract, same float math --
the entire disagreement is what that one skin actually sells for.

## Two separate problems

**Realization.** To sell into a thin book you must undercut, and with no
sales history there is no evidence anyone pays the listed number.

**Concentration.** `n*` already tells you how many contracts before the edge
beats the noise -- but `n*` assumes the prices are CORRECT and only the roll
is random. It cannot see that 99.6% of your EV rests on one questionable
number. That needs its own guard.

## Tiers

Using sell listings (what bulk-warm gives you):

| listings | tier | you realize |
|---|---|---|
| < 3 | dead | 55% |
| < 15 | thin | 78% |

With 24h sales volume (from `warm-prices`) the thresholds are 1 / 5 / 25 and

**These factors are blunt.** They are not calibrated against measured fill

# CS2 Trade-Up Lab

A free, self-hosted trade-up analysis engine for Counter-Strike 2. It pulls
live Steam Community Market prices, computes the exact output float, the
outcome probability distribution, and the fee-adjusted expected value of a
contract — then searches for the best recipes you can afford on a budget.

No API key, no subscription. The only rate limit is Steam's, and the price
layer is built around not tripping it.

---

# 3. Scan

## Rate limiting: read this before fetching prices

Steam's `429 Too Many Requests` on the market is an **IP-level** limit, not
per-account. Two things make it worse than a normal rate limit:

* A ban commonly lasts **hours**.
* It **re-arms every time you hit it while banned**. Retrying keeps you locked out.

`priceoverview` returns **one item per request**, so a Mil-Spec sweep is ~5,600
requests. That is unwinnable at any pace. `bulk-warm` uses the market's own
paginated search endpoint — 100 items per request, ~56 requests for the same
data. That is the same endpoint the market web page calls, so it *is* the
efficient form of scraping; driving a browser at it would be strictly worse.

| | `priceoverview` | `bulk-warm` |
|---|---|---|
| Items per request | 1 | 100 |
| Lowest price | yes | yes |
| Median price | yes | **no** |
| 24h sales volume | yes | **no** — listing count instead |

`sell_listings` is how many are **listed right now** — supply, not sales. So the
intended workflow is two-stage:

1. `bulk-warm` prices the whole universe cheaply → `scan` shortlists recipes.
2. `warm-prices` on just those names gets true median + volume before you
   commit money — dozens of precise requests instead of thousands.

If you get banned:

```powershell
python -m tradeup.cli cooldown --currency BRL --wait
python diagnose_steam.py --currency BRL
```

A router reboot that changes your IP also clears it.

## The math it implements

### 1. Output float (normalized mapping)

The output wear is **deterministic**, not random. Each input float is
normalized inside *its own skin's* range, averaged, then projected onto the
*output skin's* range:

```
norm_i     = (float_i - min_i) / (max_i - min_i)
avg_norm   = sum(norm_i) / N
out_float  = out_min + avg_norm * (out_max - out_min)
```

The most misunderstood mechanic in trade-ups. A 0.04 float on a 0.00–0.08 skin
is **0.50** normalized; the same 0.04 on a 0.00–1.00 skin is **0.04**. Raw float
averaging is wrong whenever inputs have different ranges.

Consequences: **wide-range skins are better inputs** at the same absolute float,
and the *output's* own min/max — not your inputs' — decides the final wear
bracket. `required_avg_norm_for_wear()` inverts the formula so you can ask
"what average do I need to land this in Factory New?".

### 2. Outcome probabilities — and the honest caveat

Valve does not publish the outcome-selection rule and community calculators
disagree. Both models are implemented, and `--robust` (default on) keeps only
recipes profitable under **both**:

| Model | Rule | 5A / 5B mix (A has 2 outputs, B has 4) |
|---|---|---|
| `outcome_weighted` | `P(o in c) = n_c / sum(n_k * O_k)` | A gets 33% of the mass |
| `per_input` | pick an input (1/N), then one of its collection's outputs | A gets 50% |

Identical for single-collection contracts; they diverge sharply on
mixed-collection filler builds — exactly where "profitable" recipes appear. Any
tool that hides this is selling confidence it does not have.

### 3. Fees — the 15% is added on top, not taken out

Steam charges 5% (platform) + 10% (CS2 game fee), both computed on the
**seller's proceeds**, each with a $0.01 minimum:

```
buyer_pays  = seller_gets * 1.15
seller_gets = buyer_pays / 1.15   (about 86.96% of the listing)
```

The common `* 0.85` shortcut understates your take by ~2 percentage points and
silently kills marginal recipes. Pass `--fee 0.02` for CSFloat, `0.08` for
Skinport, etc.

### 4. Souvenir inputs (rule change, 22 May 2026)

Souvenir items are now valid trade-up inputs. All souvenir attributes are
stripped and the **output is always normal quality** — feeding ten Souvenirs in
does not produce a Souvenir out. The tool prices inputs with the `Souvenir `
prefix and *always* prices outputs without it; there is a test asserting the
output is never quoted as a Souvenir.

This matters most at Consumer and Industrial grade, where most supply comes
from souvenir-bearing tournament and map collections. Because the contract
destroys the provenance, there is no reason to pay a premium for it — souvenir
copies are often the cheapest way to fill a slot.

Guides that say souvenirs cannot enter trade-ups predate this patch. Use
`--no-souvenir` to model the old rules.

**Which skins actually have a Souvenir variant?** Souvenir and StatTrak are
mutually exclusive and come from different sources (cases vs tournament
packages), so `souvenir_capable` is derived from **crate membership**, not from
the upstream `souvenir` boolean — that flag is set on nearly every finish and
would claim ~1400 Souvenir listings exist. `sync-db` warns if the derived rate
looks implausible. To check against the live market:

```powershell
python -m tradeup.cli verify-souvenir --rarity consumer --currency BRL
```

**Pricing caveat:** a Souvenir's value is dominated by its gold stickers (event,
teams, MVP autograph), which the market hash name does not capture. Two
identical-looking Souvenirs can differ by orders of magnitude. That is tolerable
here only because souvenirs are bought as fodder at the `lowest` listing — never
read the `median` of a Souvenir name as a real valuation.

### 5. Expected value, and what it does not tell you

```
EV_net = sum(P(o) * net_price(o))
profit = EV_net - sum(input_costs)
ROI    = EV_net / cost
```

Positive EV is necessary, not sufficient. Also reported:

- **P(win)** — how often a *single* contract beats your cost. Many +EV trade-ups
  win 8% of the time: a good lottery ticket, not income.
- **sigma** — outcome standard deviation.
- **n\*** — how many identical contracts before the edge outweighs the noise at
  ~2 sigma. If `n* = 340` at R$200 each, that is a bankroll problem, not a
  strategy.
- **Unpriced mass** — probability on outcomes with no price data. Recipes above
  10% are dropped, since an unpriced outcome scores as zero.

### 6. Price realism

The default policy is deliberately pessimistic, because the classic way these
calculators lie is using `lowest_price` on both sides:

- **Inputs** at `lowest` — what you actually pay.
- **Outputs** at `median` — the lowest listing is often one outlier you cannot
  reproduce ten times.
- **Haircut** (3%) on outputs.
- **Output volume floor** (`--min-volume`, default 3).
- **Input volume floor** (`--min-input-volume`, default 10) — you need *ten*
  copies. Souvenir listings are often near-unique, so a R$0.40 quote backed by
  two daily sales is not a real price.
- **Float percentile** (0.15) — "with some hunting I can source an item in the
  bottom 15% of this wear bracket". Use `0.0` only if you will pay float-sniper
  premiums; `0.5` for a lazy market-order assumption.
